\begin{circuitikz}[american, thick, transform shape, circuit ee IEC]

  \input{tikz/ckt_lib.tex}

  % Layout constants
  \def\wy{3.2}     % main wire y-level
  \def\mid{1.6}    % intermediate y (V_S bottom / Z_g top)
  \newcommand{\widehimpedance}[1]{
    ++(0,0) coordinate (cStart)
    -- ++(0.6,0)
    -- ++(0,0.25)
    -- ++(0.9,0)
    -- ++(0,-0.5)
    -- ++(-0.9,0)
    -- ++(0,0.25)
    ++(0.9,0)
    -- ++(0.6,0)
    coordinate (impEnd)
    ++(-1.05,0)
    node [anchor=center] {#1}
    (impEnd)
  }

  % ----- Source side -----
  % Z_g_src from ground up to V_S(-) node
  \draw (0,0) \vground;
  \draw (0,0) to [generic, l=$Z_g$] (0,\mid);

  % Voltage source from V_S(-) up to main wire — flipped
  \draw (0,\mid) to [V, l_=$V_S$, invert] (0,\wy);

  % Source impedance Z_src horizontal
  \draw (0,\wy) \widehimpedance{$Z_{src}$} coordinate (pi_left);

  % ----- PI wire model -----
  % Left shunt C/2
  \fill (pi_left) circle (2pt);
  \draw (pi_left) to [C, l=$\frac{C}{2}$] ($(pi_left)+(0,-\wy)$);
  \draw ($(pi_left)+(0,-\wy)$) \vground;

  % Series impedance Z_w
  \draw (pi_left) \widehimpedance{$Z_w$} coordinate (pi_right);

  % Right shunt C/2
  \fill (pi_right) circle (2pt);
  \draw (pi_right) to [C, l=$\frac{C}{2}$] ($(pi_right)+(0,-\wy)$);
  \draw ($(pi_right)+(0,-\wy)$) \vground;

  % ----- Destination side -----
  % Destination impedance Z_dst horizontal
  \draw (pi_right) \widehimpedance{$Z_{dst}$} coordinate (dst);

  % Open terminal V_ref,+ at destination top
  \draw (dst) to [short, -o] ++(0.8,0) node[anchor=west] {$V_{ref,+}$};

  % Define Z_g top coordinate (no wire connecting to V_ref+)
  \coordinate (dst_zg_top) at ($(dst)+(0,-\mid)$);

  % Open terminal V_ref,- at top of Z_g
  \draw (dst_zg_top) to [short, -o] ++(0.8,0) node[anchor=west] {$V_{ref,-}$};

  % Z_g from \mid down — aligned with left Z_g
  \draw (dst_zg_top) to [generic, l=$Z_g$] ($(dst_zg_top)+(0,-\mid)$);
  \draw ($(dst_zg_top)+(0,-\mid)$) \vground;

\end{circuitikz}
