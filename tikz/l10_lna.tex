% The low power LNA: a complementary common source stage.  The antenna
% comes in through two AC coupling capacitors, the gates are biased
% through resistors from V_gp and V_gn, and the drains drive the mixer.

\begin{circuitikz}[american, thick, transform shape, circuit ee IEC]

  \input{tikz/ckt_lib.tex}

  \def\xbias{3.15}
  \def\yout{2.9}
  \def\ygp{4.6}
  \def\ygn{1.2}

  % the two devices
  \draw (4.5,0) \vground;
  \draw (4.5,0) -- (4.5,0.4);
  \draw (4.5,0.4) \lvnmos{MN}{};
  \draw (4.5,2.0) -- (4.5,3.8);
  \draw (4.5,3.8) \lvpmos{MP}{};
  \draw (4.5,5.4) \vsupply;

  % the input, split to the two coupling capacitors
  \draw (0.45,\yout) -- (1.2,\yout);
  \draw (0.36,\yout) circle (0.09);
  \node[anchor=east] at (0.2,\yout) {Antenna};
  \node[anchor=east] at (0.2,\yout-0.45) {/Match};
  \draw (1.2,\ygn) -- (1.2,\ygp);

  \draw (1.4,\ygp) \hcapacitor{};
  \draw (2.9,\ygp) -- (4.0,\ygp);
  \fill (\xbias,\ygp) circle (0.075);
  \draw (1.4,\ygn) \hcapacitor{};
  \draw (2.9,\ygn) -- (4.0,\ygn);
  \fill (\xbias,\ygn) circle (0.075);
  \draw (1.2,\ygp) -- (1.4,\ygp);
  \draw (1.2,\ygn) -- (1.4,\ygn);

  % The gate bias resistors. They run vertically, away from the device
  % they bias, so that nothing has to cross the coupling capacitors and
  % the two bias nodes end up at the same height as the rails they sit
  % next to. Drawn diagonally, as in the original, they read as if they
  % were shorting the two gates together.
  \draw (\xbias,\ygp) \vresistor{};
  \draw (\xbias,6.2) to [short,-o] (\xbias,6.32);
  \node[anchor=south] at (\xbias,6.45) {$V_{gp}$};
  \draw (\xbias,-0.4) \vresistor{};
  \draw (\xbias,-0.4) to [short,-o] (\xbias,-0.52);
  \node[anchor=north] at (\xbias,-0.68) {$V_{gn}$};

  % the output to the mixer
  \draw (4.5,\yout) -- (5.5,\yout);
  \fill (4.5,\yout) circle (0.075);
  \draw (5.59,\yout) circle (0.09);
  \node[anchor=west] at (5.75,\yout) {Mixer};

\end{circuitikz}
\end{document}
