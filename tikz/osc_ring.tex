% Ring oscillator: three inverters in a loop, each with a propagation
% delay t_pd, and a fourth inverter as output buffer.

\begin{circuitikz}[american, thick]

  \node[not port] (g1) at (0,0) {};
  \node[not port] (g2) at (2.2,0) {};
  \node[not port] (g3) at (4.4,0) {};
  \node[not port] (g4) at (6.9,0) {};

  \draw (g1.out) -- (g2.in);
  \draw (g2.out) -- (g3.in);
  \draw (g3.out) to [short,-*] (5.65,0) -- (g4.in);
  \draw (g4.out) -- ++(0.5,0);

  % feedback
  \draw (5.65,0) -- (5.65,-1.4) -- (-1.4,-1.4) -- (-1.4,0) -- (g1.in);

  \node[anchor=south] at (4.4,0.65) {$t_{pd}$};

\end{circuitikz}
