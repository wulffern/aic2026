% Nauta's transconductor: two inverters amplify, and four small
% inverters hold the common mode - a cross coupled pair between the
% outputs, and a shorted inverter load on each output. Everything is an
% inverter, so the OTA works at any supply where an inverter has gain.

\begin{circuitikz}[american, thick, transform shape]

  \def\yt{1.6}      % top signal path
  \def\yb{-1.6}     % bottom signal path

  % main transconductors
  \draw (-3.4,\yt) node[anchor=east] {$v_p$} to [short,o-] (-2.0,\yt);
  \draw (-2.0,\yt) node[not port, anchor=in 1] (inv1) {};
  \draw (-3.4,\yb) node[anchor=east] {$v_n$} to [short,o-] (-2.0,\yb);
  \draw (-2.0,\yb) node[not port, anchor=in 1] (inv2) {};

  % output rails
  \draw (inv1.out) -- (5.2,\yt);
  \draw (5.2,\yt) to [short,-o] ++(0.5,0) node[anchor=west] {$v_{on}$};
  \draw (inv2.out) -- (5.2,\yb);
  \draw (5.2,\yb) to [short,-o] ++(0.5,0) node[anchor=west] {$v_{op}$};

  % cross coupled pair between the outputs
  \draw (0.6,\yt) node[not port, scale=0.7, rotate=-90, anchor=in 1] (inv3) {};
  \draw (inv3.out) -- (0.6,\yb);
  \fill (0.6,\yt) circle (0.075);
  \fill (0.6,\yb) circle (0.075);

  \draw (1.8,\yb) node[not port, scale=0.7, rotate=90, anchor=in 1] (inv4) {};
  \draw (inv4.out) -- (1.8,\yt);
  \fill (1.8,\yt) circle (0.075);
  \fill (1.8,\yb) circle (0.075);

  % shorted inverter load on each output
  \draw (3.2,\yt) -- (3.2,{\yt+0.8});
  \draw (3.2,{\yt+0.8}) node[not port, scale=0.7, anchor=in 1] (inv5) {};
  \draw (inv5.out) -- (4.6,{\yt+0.8}) -- (4.6,\yt);
  \fill (3.2,\yt) circle (0.075);
  \fill (4.6,\yt) circle (0.075);

  \draw (3.2,\yb) -- (3.2,{\yb-0.8});
  \draw (3.2,{\yb-0.8}) node[not port, scale=0.7, anchor=in 1] (inv6) {};
  \draw (inv6.out) -- (4.6,{\yb-0.8}) -- (4.6,\yb);
  \fill (3.2,\yb) circle (0.075);
  \fill (4.6,\yb) circle (0.075);

\end{circuitikz}
