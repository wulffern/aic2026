% Nauta's transconductor: two inverters amplify, and four small
% inverters hold the common mode - a cross coupled pair between the
% outputs, and a shorted inverter load on each output. Everything is an
% inverter, so the OTA works at any supply where an inverter has gain.

\begin{circuitikz}[american, thick, transform shape, circuit ee IEC]

  \input{tikz/ckt_lib.tex}

  \def\yt{1.6}      % top signal path
  \def\yb{-1.6}     % bottom signal path

  % main transconductors
  \draw (-2.0,\yt) \cicInv{inv1};
  \draw (-3.4,\yt) node[anchor=east] {$v_p$} to [short,o-] (inv1_in);
  \draw (-2.0,\yb) \cicInv{inv2};
  \draw (-3.4,\yb) node[anchor=east] {$v_n$} to [short,o-] (inv2_in);

  % output rails
  \draw (inv1_out) -- (5.2,\yt);
  \draw (5.2,\yt) to [short,-o] ++(0.5,0) node[anchor=west] {$v_{on}$};
  \draw (inv2_out) -- (5.2,\yb);
  \draw (5.2,\yb) to [short,-o] ++(0.5,0) node[anchor=west] {$v_{op}$};

  % cross coupled pair between the outputs, small
  \begin{scope}[shift={(0.6,\yt)}, rotate=-90, scale=0.8]
    \draw (0,0) \cicInv{inv3};
  \end{scope}
  \draw (inv3_out) -- (0.6,\yb);
  \fill (0.6,\yt) circle (0.075);
  \fill (0.6,\yb) circle (0.075);

  \begin{scope}[shift={(1.8,\yb)}, rotate=90, scale=0.8]
    \draw (0,0) \cicInv{inv4};
  \end{scope}
  \draw (inv4_out) -- (1.8,\yt);
  \fill (1.8,\yt) circle (0.075);
  \fill (1.8,\yb) circle (0.075);

  % shorted inverter load on each output, small
  \draw (3.2,\yt) -- (3.2,{\yt+0.8});
  \begin{scope}[shift={(3.2,{\yt+0.8})}, scale=0.8]
    \draw (0,0) \cicInv{inv5};
  \end{scope}
  \draw (inv5_out) -- (4.6,{\yt+0.8}) -- (4.6,\yt);
  \fill (3.2,\yt) circle (0.075);
  \fill (4.6,\yt) circle (0.075);

  \draw (3.2,\yb) -- (3.2,{\yb-0.8});
  \begin{scope}[shift={(3.2,{\yb-0.8})}, scale=0.8]
    \draw (0,0) \cicInv{inv6};
  \end{scope}
  \draw (inv6_out) -- (4.6,{\yb-0.8}) -- (4.6,\yb);
  \fill (3.2,\yb) circle (0.075);
  \fill (4.6,\yb) circle (0.075);

\end{circuitikz}
