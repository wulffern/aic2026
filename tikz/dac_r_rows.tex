% The matrix decode, drawn as the circuit it is rather than as abstract
% knife switches: sixteen taps from the string enter a 4x4 array of pass
% devices, a row-select line gates each rank of four, and one column
% device per column line joins the shared output rail. One switch per tap
% plus one per column is the 2^N + 2^(N/2) the lecture counts, against the
% tree's 2^(N+1) - 2. The devices are the same NMOS-with-control the
% string figures (dac_r_div*) use, so the two slides read as one family.

\begin{circuitikz}[american, thick, transform shape, circuit ee IEC]

  \input{tikz/ckt_lib.tex}

  \def\cx{2.6}       % column pitch
  \def\ry{1.9}       % row pitch

  % ---- The 4x4 array: tap port, pass device, row-select line ----
  \foreach \r in {0,1,2,3} {
    \foreach \c in {0,1,2,3} {
      \pgfmathtruncatemacro{\tap}{4*\r+\c}
      \draw ({\c*\cx},{\r*\ry}) \portIn{$V_{\tap}$}
            -- ++(0.36,0) to [Tnmos, n=M\c\r] ++(1.0,0)
            -- ++(0.3,0) coordinate (o\c\r);
    }
    % The row line runs through all four gates; the interior gates are
    % T junctions, the two end gates are where the line starts and stops.
    \draw (M0\r.gate) -- (M3\r.gate);
    \fill (M1\r.gate) circle (0.07);
    \fill (M2\r.gate) circle (0.07);
    \draw (M0\r.gate) -- ++(-1.6,0) node[anchor=east] {$r_\r$};
  }

  % ---- Column rails and the one switch per column ----
  % The rail crosses the row lines below its own cell without a dot:
  % crossing is not connection.
  \foreach \c in {0,1,2,3} {
    \draw (o\c3) -- (o\c0) -- ++(0,-0.6) coordinate (ctop\c);
    \foreach \r in {0,1,2} { \fill (o\c\r) circle (0.07); }
    \draw (ctop\c) to [Tnmos, n=MC\c] ++(0,-1.2) -- ++(0,-0.4) coordinate (cbot\c);
    \draw (MC\c.gate) -- ++(0.35,0) node[anchor=west] {$c_\c$};
  }

  % ---- The shared output rail ----
  \draw (cbot0) -- (cbot3);
  \fill (cbot1) circle (0.07);
  \fill (cbot2) circle (0.07);
  \fill (cbot3) circle (0.07);
  \draw (cbot3) -- ++(0.7,0) coordinate (vo);
  \draw (vo) circle (0.06);
  \node[anchor=west, xshift=3pt] at (vo) {$V_{OUT}$};

\end{circuitikz}
\end{document}
