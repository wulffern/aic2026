% The clocks of a wireless SoC (an nRF-style IC): a 32 MHz crystal
% oscillator feeds the system PLL, which clocks the MCU; the radio has
% its own PLL and LO buffer. The slow 32768 Hz clock comes from either
% a crystal oscillator or an RC oscillator.

\begin{circuitikz}[american, thick,
  blk/.style={draw, rectangle, minimum width=1.3cm,
              minimum height=1.0cm, text=red}]

  % Crystal symbol at (#1,#2), drawn upright: plate, box, plate, with a
  % pin out of the top and the bottom. Both pins belong to the same
  % oscillator - a crystal is a two terminal device across the XO.
  \newcommand{\xtal}[2]{%
    \draw (#1,{#2+0.8}) -- (#1,{#2+0.36});
    \draw ({#1-0.42},{#2+0.36}) -- ({#1+0.42},{#2+0.36});
    \draw ({#1-0.24},{#2-0.26}) rectangle ({#1+0.24},{#2+0.26});
    \draw ({#1-0.42},{#2-0.36}) -- ({#1+0.42},{#2-0.36});
    \draw (#1,{#2-0.36}) -- (#1,{#2-0.8});}

  % the IC boundary
  \draw (0,0) rectangle (7.0,7.6);

  % 32 MHz crystal oscillator and system PLL
  \node[blk] (xo1) at (1.3,6.6) {XO};
  \node[blk] (pll) at (1.3,5.3) {PLL};
  \node[blk, minimum width=2.4cm] (mcu) at (4.4,3.4) {MCU};

  % the radio with its own PLL and the LO buffer
  \draw (3.3,4.6) rectangle (6.3,6.3);
  \node[red,anchor=north] at (4.35,6.25) {\small RADIO};
  \node[blk, minimum width=1.0cm, minimum height=0.8cm] (rpll) at (4.2,5.3) {\small PLL};
  \draw (5.0,4.9) -- (5.0,5.7) -- (5.7,5.3) -- cycle;   % the LO buffer
  \node[armygreen, anchor=south] at (5.28,5.72) {\small LO};

  % antenna, outside the chip
  \draw (5.7,5.3) -- (8.0,5.3) -- (8.0,8.4);
  \draw (8.0,8.4) -- (7.6,9.0) (8.0,8.4) -- (8.4,9.0) (7.6,9.0) -- (8.4,9.0);

  % 32 MHz crystal, outside the chip, both pins into the XO
  \xtal{-1.5}{6.6}
  \draw (-1.5,7.4) -- (-1.5,7.75) -- (0.65,7.75) -- (0.65,6.95);
  \draw (-1.5,5.8) -- (-1.5,5.45) -- (0.65,5.45) -- (0.65,6.25);

  % XO to the PLLs, and the 32 MHz label
  \draw (xo1.east) -- (2.6,6.6);
  \node[armygreen, anchor=south west] at (2.75,6.45) {32 MHz};
  \draw (2.6,6.6) -- (2.6,5.3);
  \draw[<->] (pll.east) -- (rpll.west);
  \fill (2.6,5.3) circle (0.05);

  % the system PLL clocks the MCU
  \draw[->] (pll.south) -- (1.3,3.4) -- (mcu.west);

  % 32768 Hz: crystal oscillator or RC oscillator
  \xtal{-1.5}{2.6}
  \node[blk] (xo2) at (1.3,2.6) {XO};
  \draw (-1.5,3.4) -- (-1.5,3.75) -- (0.65,3.75) -- (0.65,2.95);
  \draw (-1.5,1.8) -- (-1.5,1.45) -- (0.65,1.45) -- (0.65,2.25);
  \node[blk] (rc) at (1.3,1.0) {RC};

  % A change-over switch picks ONE of the two slow clocks: the two
  % throws stay separate, and only the wiper carries the 32768 Hz on.
  \draw (xo2.east) -- (2.5,2.6);
  \draw (rc.east) -- (2.5,1.0);
  \draw (2.5,2.6) circle (0.06);
  \draw (2.5,1.0) circle (0.06);
  \draw (3.6,1.8) circle (0.06);
  \draw (3.66,1.8) -- (4.6,1.8);          % the wiper output
  \draw (3.55,1.75) -- (2.62,2.52);       % wiper, currently on the XO
  \node[armygreen, anchor=west] at (4.65,1.8) {32768 Hz};

\end{circuitikz}
\end{document}
