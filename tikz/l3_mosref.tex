% A reference built on the difference between two threshold voltages.
% Included so it can be recognised, not so it can be copied.
%
% M1 is a depletion device with its gate tied to its source, so V_GS1 = 0 and
% it still conducts because V_t1 < 0. M2 is an ordinary enhancement device,
% diode connected. Equal currents in strong inversion give
% V_REF = V_t2 - V_t1 sqrt(W1 L2 / L1 W2), and with equal shapes V_REF is
% simply the difference of the two thresholds.
%
% Both gates sit on the same V_REF node, drawn as one bus on the left so the
% right hand side stays free for the annotations.

\begin{circuitikz}[american, thick, transform shape, circuit ee IEC]

  \input{tikz/ckt_lib.tex}

  \def\x{0}
  \def\xg{-1.1}      % gate bus
  \def\ymid{2.0}     % V_REF tap on the shared node

  % ---- M2: diode connected enhancement device ----
  \draw (\x,0) \vground;
  \draw (\x,0) \lvnmos{M2}{};

  % ---- Shared node: source of M1, drain of M2, and both gates ----
  \draw (\x,\grid) -- (\x,\grid+0.8);

  % ---- M1: depletion device, gate on its own source ----
  \draw (\x,\grid+0.8) \lvnmos{M1}{};
  \draw (\x,\grid*2+0.8) \vsupply;

  % ---- Gate bus ----
  \draw (M1.gate) -- (\xg,3.2) -- (\xg,0.8) -- (M2.gate);
  \draw (\xg,\ymid) -- (\x,\ymid);
  \fill (\xg,\ymid) circle (0.06);

  % ---- Output ----
  \draw (\x,\ymid) \portOut{$V_{REF}$};

  % ---- Annotations ----
  \node[anchor=east] at (\xg-0.2,3.2) {$V_{GS1}=0$};
  \node[anchor=west] at (\x+0.55,3.2) {$M_{1}$: depletion, $V_{t1}<0$};
  \node[anchor=west] at (\x+0.55,0.8) {$M_{2}$: enhancement, $V_{t2}>0$};

\end{circuitikz}
\end{document}
