% Switched capacitor CMFB. C1/C1 sense the average of the two outputs
% and level shift it onto v_cmfb; the switched C2s refresh the level
% shift towards V_cm - V_B every phi_1.

\begin{circuitikz}[american, thick, transform shape]

  \def\yr{0}        % sense rail
  \def\yc{2.2}      % switched cap row
  \def\yref{4.0}    % reference rail

  % sense rail: vop - C1 - vcmfb - C1 - von
  \draw (-5.6,\yr) node[anchor=east] {$v_{op}$} to [short,o-] (-4.6,\yr);
  \draw (-4.6,\yr) to [C, l=$C_1$] (-2.4,\yr);
  \draw (-2.4,\yr) -- (2.4,\yr);
  \draw (2.4,\yr) to [C, l=$C_1$] (4.6,\yr);
  \draw (4.6,\yr) to [short,-o] (5.6,\yr) node[anchor=west] {$v_{on}$};
  \fill (0,\yr) circle (0.075);
  \draw (0,\yr) to [short,-o] (0,-1.0) node[anchor=north] {$v_{cmfb}$};

  % left switched capacitor
  \draw (-4.6,\yr) to [nos, l=$\phi_2$] (-4.6,\yc);
  \fill (-4.6,\yr) circle (0.075);
  \draw (-4.6,\yc) to [C, l=$C_2$] (-2.4,\yc);
  \draw (-2.4,\yc) to [nos, l_=$\phi_2$] (-2.4,\yr);
  \fill (-2.4,\yr) circle (0.075);
  \draw (-4.6,\yc) to [nos, l=$\phi_1$] (-4.6,\yref);
  \fill (-4.6,\yc) circle (0.075);
  \draw (-4.6,\yref) to [short,-o] ++(-0.4,0) node[anchor=east] {$V_{cm}$};
  \draw (-2.4,\yc) to [nos, l_=$\phi_1$] (-2.4,\yref);
  \fill (-2.4,\yc) circle (0.075);
  \draw (-2.4,\yref) to [short,-o] ++(0.4,0) node[anchor=west] {$V_{B}$};

  % right switched capacitor
  \draw (4.6,\yr) to [nos, l_=$\phi_2$] (4.6,\yc);
  \fill (4.6,\yr) circle (0.075);
  \draw (4.6,\yc) to [C, l_=$C_2$] (2.4,\yc);
  \draw (2.4,\yc) to [nos, l=$\phi_2$] (2.4,\yr);
  \fill (2.4,\yr) circle (0.075);
  \draw (4.6,\yc) to [nos, l_=$\phi_1$] (4.6,\yref);
  \fill (4.6,\yc) circle (0.075);
  \draw (4.6,\yref) to [short,-o] ++(0.4,0) node[anchor=west] {$V_{cm}$};
  \draw (2.4,\yc) to [nos, l=$\phi_1$] (2.4,\yref);
  \fill (2.4,\yc) circle (0.075);
  \draw (2.4,\yref) to [short,-o] ++(-0.4,0) node[anchor=east] {$V_{B}$};

\end{circuitikz}
