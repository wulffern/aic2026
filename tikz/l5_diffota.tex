% The fully differential two-stage OTA, one side drawn, redrawn from
% media/diff_ota.png. Sizes are "WFLF": 24F4F means W = 24 and L = 4
% minimum gate lengths. A cascoded PMOS tail feeds the input pair, a
% cascoded mirror makes the first stage output, and the second stage
% is a common source with 500f of cascode compensation. The common
% mode correction arrives as V_CMFB from the CMFB amplifier, which
% has its own figure.

\begin{circuitikz}[american, thick, transform shape, circuit ee IEC]

  \input{tikz/ckt_lib.tex}

  \def\ytop{6.4}
  \newcommand{\dsize}[3]{\node[red, scale=0.62, anchor=west, align=left]
    at (#1,#2) {#3};}

  % ---- rails ----
  \draw[line width=2pt] (7.0,\ytop) -- (21.8,\ytop);
  \draw[line width=2pt] (7.0,0) -- (21.8,0);
  \node[anchor=west, scale=0.8] at (21.85,\ytop) {$VDD$};
  \node[anchor=west, scale=0.8] at (21.85,0) {$VSS$};

  % ================= first stage =================
  % cascoded tail: VBP on the rail, VCP below it
  \draw (10.0,4.8) \lvpmos{tp1}{$VBP$};
  \dsize{10.15}{5.9}{24F4F\\M=2}
  \draw (10.0,3.2) \lvpmos{tp2}{$VCP$};
  \dsize{10.15}{4.3}{24F2F\\M=4}

  % PMOS input pair, sources at the cascode drain
  \draw (8.8,3.2) -- (11.2,3.2);
  \fill (10.0,3.2) circle (0.075);
  \draw (8.8,1.6) to [Tpmos, n=min] ++(0,1.6);
  \draw (8.8,1.6) to [short,-o] (8.8,1.2);
  \draw (min.gate) \portIn{$VIN$};
  \dsize{8.95}{2.0}{24F1F25\\M=10}
  \draw (11.2,1.6) to [Tpmos, n=mip, mirror] ++(0,1.6);
  \draw (mip.gate) \portmIn{$VIP$};
  \dsize{10.05}{2.0}{24F1F25\\M=10}
  \node[scale=0.6, anchor=north] at (10.0,0.85) {Only one side of OTA drawn};

  % The drawn side: the VIP drain, the diode's own gate and its drain are
  % one node, so they meet at the drain and share one dot. Tapping them
  % at 1.3 and 1.45 instead put two wires and two dots on the device's
  % drain lead, a tenth of a unit apart.
  \draw (12.8,0) to [Tnmos, n=nd] ++(0,\grid);
  \draw (11.2,1.6) -- (12.8,1.6);
  \fill (12.8,1.6) circle (0.075);
  % The diode connection drops to the shared gate rail on the free side
  % of the device. Taking it over the top ran the wire under the input
  % pair and straight through the VCN label.
  \draw (12.8,1.6) -- (13.3,1.6) -- (13.3,0.8);
  \fill (13.3,0.8) circle (0.075);
  \dsize{12.95}{0.45}{24F4F\\M=1}

  % ================= cascoded load and mirror =================
  % top PMOS pair: one VBP biased, one driven by the CMFB
  \draw (14.0,4.8) \lvpmos{lp1}{$VBP$};
  \dsize{14.15}{5.9}{24F4F\\M=1}
  \draw (15.6,4.8) to [Tpmos, n=pcm, mirror] ++(0,\grid);
  \dsize{16.75}{6.0}{24F4F\\M=1}
  \draw (14.0,4.8) -- (15.6,4.8);
  \fill (14.8,4.8) circle (0.075);
  % the CMFB correction arrives as a net, not a wire over the rail
  %- down into the clear space between the mirror and the output
  %- branch, so it does not fight the VBP1 label
  \draw (16.58,5.6) -- (17.4,5.6) -- (17.4,4.3) to [short,-o] ++(0.35,0);
  \node[anchor=west, scale=0.85] at (17.85,4.3) {$V_{CMFB}$};

  % VCP cascode down to the first stage output X
  \draw (14.8,3.2) \lvpmos{lp2}{$VCP$};
  \dsize{14.95}{4.3}{24F2F\\M=4}
  % X is where the two cascodes meet; tap it there rather than a
  % fraction of a unit down inside the NMOS.
  \fill (14.8,3.2) circle (0.075);

  % VCN cascode and the mirror NMOS below it
  \draw (14.8,1.6) \lvnmos{ln1}{};
  \draw (13.82,2.4) -- (13.4,2.4) -- (13.4,1.95) \portIn{$VCN$};
  \dsize{14.95}{2.0}{24F2F\\M=8}
  \draw (14.8,0) to [Tnmos, n=nm] ++(0,\grid);
  \dsize{14.95}{0.45}{24F4F\\M=2}
  \draw (13.82,0.8) -- (11.82,0.8);
  \fill (14.8,1.6) circle (0.075);

  % ================= second stage =================
  % X drives the output NMOS gate; the vertical crosses the 500f
  % wire without connecting, as in the original
  \draw (14.8,3.2) -- (16.4,3.2) -- (16.4,0.8) -- (19.62,0.8);
  % output branch: VBP1 source, VON, output NMOS
  \draw (20.6,4.8) \lvpmos{op1}{$VBP1$};
  \dsize{20.75}{5.9}{24F4F\\M=4}
  \draw (20.6,4.8) -- (20.6,3.6);
  \draw (20.6,3.6) to [short,-o] ++(0.5,0) node[anchor=west] {$VON$};
  \fill (20.6,3.6) circle (0.075);
  \draw (20.6,3.6) -- (20.6,1.6);
  \draw (20.6,0) to [Tnmos, n=no] ++(0,\grid);
  \dsize{19.5}{0.45}{24F2F\\M=2}
  % 500f cascode compensation from the internal node to VON
  \fill (20.6,1.6) circle (0.075);
  \draw (14.8,1.6) -- (19.0,1.6) \hcapacitor{500f} -- (20.6,1.6);


\end{circuitikz}
