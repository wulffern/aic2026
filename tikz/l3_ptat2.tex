% Banba-style combination: each branch carries a diode leg and an R2 leg in
% parallel off the same node, so the PMOS current is I_1 + I_2 with
% I_1 = dV_BE/R1 (PTAT, through R1) and I_2 = V_BE1/R2 (CTAT, through R2).
% Four legs left to right: R2, unit diode, R1 + xN diode, R2.
%
% OTA polarity corrected as in l3_ptat — the artwork marks + on the left, but
% the right-hand node is the one that rises with current through R1.

\begin{circuitikz}[american, thick, transform shape, circuit ee IEC]

  \input{tikz/ckt_lib.tex}

  \def\xrl{-\grid*1.5}   % left R2 leg
  \def\xl{0}             % unit diode branch
  \def\xr{\grid*2.5}     % R1 + xN diode branch
  % The right R2 leg is pushed out to leave room for the I_1 expression, which
  % sits between it and the R1 branch as in the original.
  \def\xrr{\grid*5}      % right R2 leg
  \def\ynode{3.0}
  \def\pdlevel{5}

  % ---- Diode legs ----
  \node[pnp, anchor=C] (Q1) at (\xl, 0) {};
  \draw (Q1.C) -- ++(0,-0.2) \vground;
  \draw (Q1.B) |- (Q1.C);
  \draw (Q1.E) -- (\xl,\pdlevel);

  \node[pnp, anchor=C, label=right:${\times N}$] (Q2) at (\xr, 0) {};
  \draw (Q2.C) -- ++(0,-0.2)\vground;
  \draw (Q2.B) |- (Q2.C);
  \draw (Q2.E) \vresistor{$R_{1}$};
  \draw (resEnd) -- (\xr,\pdlevel);

  % ---- R2 legs, one per branch, tied in at the sense node ----
  \draw (\xrl,0) \vground;
  \draw (\xrl,0) -- ++(0,0.4);

  \draw (resEnd) -- (\xrl,\ynode) -- (\xl,\ynode);

  \draw (\xrr,0) \vground;
  \draw (\xrr,0) -- ++(0,0.4);
  \draw (\xrr,0.4) \vresistor{$R_{2}$};

  \node at ($(otaRight)+(0,0.3)$) {$+$};
