% The s-plane. The heavy bar is the imaginary axis, which is where the
% frequency response is evaluated, and the two poles sit in the left half
% plane where a continuous time system is stable.
%
% First of three plane diagrams; l5_zdomain and l5_zunstable follow. Geometry
% comes from tikz/plane_lib.tex so the axes line up across all three.

\input{tikz/plane_lib.tex}

\begin{circuitikz}[american, transform shape]

  \planeAxes{$a$}{$j\omega$}

  % The jw axis itself, drawn heavy: this is the line the frequency response
  % is read along, and it is what maps to the unit circle in the z-plane.
  \draw[axisblue, line width=4pt] (0,-2.2) -- (0,2.2);

  % A conjugate pole pair in the left half plane.
  \planePole{axisblue}{-0.55}{0.32}
  \planePole{axisblue}{-0.55}{-0.32}

  \node[axisblue, anchor=west] at (1.15,2.45) {$s = a + j\omega$};

\end{circuitikz}
\end{document}
