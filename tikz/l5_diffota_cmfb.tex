% The common mode feedback amplifier of the two-stage OTA, drawn on
% its own so the loop is readable. VON and VOP are sensed through
% 60k||20f networks and compared against a 100k/100k mid-supply
% reference by a differential pair. Only the LEFT load is diode
% connected; the right one mirrors it, so the amplifier's output is
% the right drain - the high impedance node - which leaves as V_CMFB
% and trims the first stage load of the OTA.

\begin{circuitikz}[american, thick, transform shape, circuit ee IEC]

  \input{tikz/ckt_lib.tex}

  \def\ytop{6.4}
  \newcommand{\dsize}[3]{\node[red, scale=0.62, anchor=west, align=left]
    at (#1,#2) {#3};}

  % ---- rails ----
  \draw[line width=2pt] (-4.0,\ytop) -- (8.0,\ytop);
  \draw[line width=2pt] (-4.0,0) -- (8.0,0);
  \node[anchor=west, scale=0.8] at (8.05,\ytop) {$VDD$};
  \node[anchor=west, scale=0.8] at (8.05,0) {$VSS$};

  % ================= CMFB amplifier =================
  % PMOS diode loads, drains joined: that node is the CMFB output
  \draw (1.0,4.8) to [Tpmos, n=cp1] ++(0,\grid);
  \draw (3.4,4.8) to [Tpmos, n=cp2] ++(0,\grid);
  % Only the LEFT device is diode connected; the right one mirrors it.
  % The amplifier's output is therefore the RIGHT drain - the high
  % impedance node - not the diode node, and it leaves as a named net
  % rather than a wire across the supply rail.
  \draw (0.02,5.6) -- (0.02,4.5) -- (1.0,4.5) -- (1.0,4.8);
  \draw (2.42,5.6) -- (2.42,4.5) -- (1.0,4.5);
  \fill (1.0,4.5) circle (0.075);
  \fill (3.4,4.4) circle (0.075);
  \draw (3.4,4.4) -- (4.3,4.4) to [short,-o] ++(0.35,0);
  \node[anchor=west, scale=0.85] at (4.75,4.4) {$V_{CMFB}$};
  \dsize{1.15}{5.9}{24F4F\\M=1}
  \dsize{2.1}{5.9}{24F4F\\M=1}

  % NMOS pair under the loads
  \draw (1.0,1.6) to [Tnmos, n=cn1] ++(0,\grid);
  \draw (3.4,1.6) to [Tnmos, n=cn2, mirror] ++(0,\grid);
  \draw (1.0,3.2) -- (1.0,4.8);
  \draw (3.4,3.2) -- (3.4,4.8);
  \dsize{1.15}{2.0}{24F1F25\\M=2}
  \dsize{2.1}{2.0}{24F1F25\\M=2}
  % common source into the VBN tail
  \draw (1.0,1.6) -- (3.4,1.6);
  \draw (2.2,0) to [Tnmos, n=ct] ++(0,\grid);
  \fill (2.2,1.6) circle (0.075);
  \draw (1.22,0.8) \portIn{$VBN$};
  \dsize{2.35}{0.45}{24F4F\\M=2}

  % VON/VOP sense networks: 20f || 60k each, joined at the gate node
  \draw (-3.4,4.75) node[anchor=south] {$VON$} to [short,o-] (-3.4,4.5);
  \draw (-3.4,2.9) \vcapacitor{};
  \draw (-3.4,4.25) -- (-2.8,4.25) -- (-2.8,4.2);
  \draw (-2.8,2.9) -- (-2.8,3.0) \vresistor{} (-2.8,4.2);
  \draw (-2.8,2.9) -- (-3.4,2.9);
  \node[scale=0.75, anchor=east] at (-3.5,3.6) {20f};
  \draw (-1.4,4.75) node[anchor=south] {$VOP$} to [short,o-] (-1.4,4.5);
  \draw (-1.4,2.9) \vcapacitor{};
  \draw (-1.4,4.25) -- (-0.8,4.25) -- (-0.8,4.2);
  \draw (-0.8,2.9) -- (-0.8,3.0) \vresistor{} (-0.8,4.2);
  \draw (-0.8,2.9) -- (-1.4,2.9);
  \node[scale=0.75, anchor=west] at (-0.65,3.6) {60k};
  \draw (-3.4,2.9) -- (-1.4,2.9);
  \fill (-1.4,2.9) circle (0.075);
  \draw (-1.4,2.9) -- (-1.4,2.4) -- (0.02,2.4);

  % 100k/100k mid-supply reference with 500f, into the right gate
  \draw (5.8,3.0) -- (5.8,3.2) \vresistor{100k} -- (5.8,\ytop);
  \fill (5.8,3.0) circle (0.075);
  \draw (5.8,0) -- (5.8,0.7) \vresistor{100k} -- (5.8,3.0);
  \draw (5.8,3.0) -- (6.5,3.0) -- (6.5,2.9);
  \draw (6.5,1.3) \vcapacitor{};
  \node[scale=0.75, anchor=east, fill=white, inner sep=1pt] at (6.35,2.1) {500f};
  \draw (6.5,1.3) -- (6.5,0);
  \draw (4.38,2.4) -- (5.8,2.4);
  \fill (5.8,2.4) circle (0.075);


\end{circuitikz}
