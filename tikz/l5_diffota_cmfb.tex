% The common mode feedback amplifier of the two-stage OTA, drawn on
% its own so the loop is readable. VON and VOP are sensed through
% 60k||20f networks and compared against a 100k/100k mid-supply
% reference by a differential pair. Only the LEFT load is diode
% connected; the right one mirrors it, so the amplifier's output is
% the right drain - the high impedance node - which leaves as V_CMFB
% and trims the first stage load of the OTA.

\begin{circuitikz}[american, thick, transform shape, circuit ee IEC]

  \input{tikz/ckt_lib.tex}

  \def\ytop{6.4}
  \newcommand{\dsize}[3]{\node[red, scale=0.62, anchor=west, align=left]
    at (#1,#2) {#3};}

  % ---- rails ----
  \draw[line width=2pt] (-4.0,\ytop) -- (9.6,\ytop);
  \draw[line width=2pt] (-4.0,0) -- (9.6,0);
  \node[anchor=west, scale=0.8] at (9.65,\ytop) {$VDD$};
  \node[anchor=west, scale=0.8] at (9.65,0) {$VSS$};

  % ================= CMFB amplifier =================
  % Each load PMOS is diode connected on its own - the two gates are
  % NOT tied together, so this is a pair of diode loads and not a
  % mirror. The gain is gm_n/gm_p, modest and well defined, which is
  % what a common mode loop wants. V_CMFB leaves from the right drain.
  \draw (1.0,4.8) to [Tpmos, n=cp1] ++(0,\grid);
  \draw (3.4,4.8) to [Tpmos, n=cp2, mirror] ++(0,\grid);
  \dsize{1.15}{5.9}{24F4F\\M=1}
  \dsize{2.35}{5.9}{24F4F\\M=1}

  %- left device: gate out, down, and back to its own drain
  \draw (0.02,5.6) -- (-0.55,5.6) -- (-0.55,4.4) -- (1.0,4.4);
  \fill (1.0,4.4) circle (0.075);
  %- right device: the same, opening the other way
  \draw (4.38,5.6) -- (4.95,5.6) -- (4.95,4.4) -- (3.4,4.4);
  \fill (3.4,4.4) circle (0.075);

  %- the correction leaves from the right drain
  \draw (3.4,4.05) -- (5.6,4.05) to [short,-o] ++(0.35,0);
  \fill (3.4,4.05) circle (0.075);
  \node[anchor=west, scale=0.85] at (6.05,4.05) {$V_{CMFB}$};

  % NMOS pair under the loads
  \draw (1.0,1.6) to [Tnmos, n=cn1] ++(0,\grid);
  \draw (3.4,1.6) to [Tnmos, n=cn2, mirror] ++(0,\grid);
  \draw (1.0,3.2) -- (1.0,4.8);
  \draw (3.4,3.2) -- (3.4,4.8);
  \dsize{1.15}{2.0}{24F1F25\\M=2}
  \dsize{2.1}{2.0}{24F1F25\\M=2}
  % common source into the VBN tail
  \draw (1.0,1.6) -- (3.4,1.6);
  \draw (2.2,0) to [Tnmos, n=ct] ++(0,\grid);
  \fill (2.2,1.6) circle (0.075);
  \draw (1.22,0.8) \portIn{$VBN$};
  \dsize{2.35}{0.45}{24F4F\\M=2}

  % VON/VOP sense networks: 20f || 60k each, joined at the gate node.
  % Both branches span the same two rows, so the parallel pair reads as
  % a parallel pair: the resistor used to run 0.3 past the junction and
  % hang there.
  \newcommand{\sense}[2]{
    \draw (#1,4.75) node[anchor=south] {#2} to [short,o-] (#1,4.35);
    \draw (#1,2.75) \vcapacitor{};
    \draw ({#1+0.6},2.75) \vresistor{};
    \draw (#1,4.35) -- ({#1+0.6},4.35);
    \draw (#1,2.75) -- ({#1+0.6},2.75);
    \fill (#1,4.35) circle (0.075);
    \fill (#1,2.75) circle (0.075);
  }
  \sense{-3.4}{$VON$}
  \sense{-1.4}{$VOP$}
  %- one branch carries the values; the other is identical
  \node[scale=0.75, anchor=east] at (-3.75,3.5) {20f};
  \node[scale=0.75, anchor=west] at (-2.45,3.5) {60k};
  \draw (-2.8,2.75) -- (-1.4,2.75);
  \fill (-2.8,2.75) circle (0.075);
  \draw (-0.8,2.75) -- (-0.8,2.4) -- (0.02,2.4);
  \fill (-0.8,2.75) circle (0.075);

  % 100k/100k mid-supply reference with 500f, into the right gate
  % One junction, not two: the capacitor and the gate wire leave the
  % divider's mid node at the same height. The capacitor branch sits far
  % enough right to clear the lower resistor's own label.
  \draw (7.4,2.4) -- (7.4,3.2) \vresistor{100k} -- (7.4,\ytop);
  \draw (7.4,0) -- (7.4,0.9) \vresistor{100k} -- (7.4,2.4);
  \draw (7.4,2.4) -- (8.8,2.4) -- (8.8,2.15);
  \draw (8.8,0.65) \vcapacitor{};
  \node[scale=0.75, anchor=west] at (8.95,1.4) {500f};
  \draw (8.8,0.65) -- (8.8,0);
  \draw (4.38,2.4) -- (7.4,2.4);
  \fill (7.4,2.4) circle (0.075);


\end{circuitikz}
\end{document}
