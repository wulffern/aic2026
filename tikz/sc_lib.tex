%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%% Shared pieces for the l05_sc switched-capacitor amplifier figures.
%%
%% l5_scintro1 and l5_scintro2 are the same circuit in its two clock phases,
%% and the lecture's whole charge-conservation argument works by comparing
%% them. They therefore have to be drawn on identical geometry, which is what
%% \scIntroFrame is for: it draws everything the two share and leaves the two
%% switch positions to the caller, since those are exactly what differs.
%%
%% Every name is prefixed: short generic names collide with LaTeX built-ins.
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\definecolor{axisblue}{rgb}{0.16,0.16,0.55}
\definecolor{capgreen}{rgb}{0.00,0.45,0.20}
\definecolor{polred}{rgb}{0.85,0.22,0.10}

\newcommand{\scYgnd}{0}       % ground rail
\newcommand{\scYin}{1.8}      % OTA inverting input, and the C1 level
\newcommand{\scXota}{4.6}     % OTA input x
\newcommand{\scXc2}{4.0}      % where the C2 branch leaves the input node
\newcommand{\scYc2}{3.8}      % C2 level
\newcommand{\scYsh}{4.8}      % the shorting path over C2
\newcommand{\scXshr}{6.6}     % right leg of the shorting path
\newcommand{\scXout}{7.6}     % C2 return leg

% A switch, drawn as a hinged blade so open and closed read at a glance.
% #1 is the left end x, #2 the y; both span 1.0.
\newcommand{\scSwOpen}[2]{%
  \draw[axisblue, thick] (#1,#2) -- (#1+0.15,#2);
  \fill[axisblue] (#1+0.15,#2) circle (0.055);
  \draw[axisblue, thick] (#1+0.15,#2) -- (#1+0.88,#2+0.48);
  \fill[axisblue] (#1+0.85,#2) circle (0.055);
  \draw[axisblue, thick] (#1+0.85,#2) -- (#1+1.0,#2);
}
\newcommand{\scSwClosed}[2]{\draw[axisblue, thick] (#1,#2) -- (#1+1.0,#2);}

% Everything l5_scintro1 and l5_scintro2 have in common. The caller fills the
% C1-to-OTA gap (2.8..3.8 at \scYin) and the short over C2 (4.8..5.8 at
% \scYsh) with either \scSwClosed or \scSwOpen.
\newcommand{\scIntroFrame}{%
  % C1, plus its return along the ground rail
  \draw[axisblue, thick] (0.6,\scYin) \hcapacitor{};
  \draw[axisblue, thick] (0.6,\scYin) -- (0.0,\scYin) -- (0.0,\scYgnd) -- (3.4,\scYgnd);
  \draw[axisblue, thick] (2.2,\scYin) -- (2.8,\scYin);
  \draw[axisblue, thick] (3.8,\scYin) -- (\scXota,\scYin);
  \node[capgreen, anchor=north] at (1.4,\scYin-0.28) {$C_{1}$};
  \node[polred, anchor=east]    at (0.5,\scYin+0.42) {$+$};
  \node[polred, anchor=west]    at (2.3,\scYin+0.42) {$-$};

  % OTA, inverting input on top as in the original
  \draw (\scXota,\scYin) \cicOtaSSWP{$-$}{$+$};
  \draw[axisblue, thick] (cicOtaS_inn) -- (3.4,\scYin-0.8) -- (3.4,\scYgnd);
  \draw[axisblue, thick] (2.6,\scYgnd) \vground;

  % C2 across the OTA
  \draw[axisblue, thick] (\scXc2,\scYin) -- (\scXc2,\scYc2);
  \fill[axisblue] (\scXc2,\scYin) circle (0.07);
  \draw[axisblue, thick] (\scXc2,\scYc2) \hcapacitor{};
  \draw[axisblue, thick] (5.6,\scYc2) -- (\scXout,\scYc2) -- (\scXout,\scYin-0.4);
  \draw[axisblue, thick] (\scXout,\scYin-0.4) -- (cicOtaS_out);
  \node[capgreen, anchor=south] at (5.8,\scYc2+0.15) {$C_{2}$};
  \node[polred, anchor=east]    at (3.9,\scYc2-0.45) {$-$};
  \node[polred, anchor=west]    at (5.7,\scYc2-0.45) {$+$};

  % The path that shorts C2, minus its switch
  \draw[axisblue, thick] (\scXc2,\scYc2) -- (\scXc2,\scYsh) -- (4.8,\scYsh);
  \draw[axisblue, thick] (5.8,\scYsh) -- (\scXshr,\scYsh) -- (\scXshr,\scYc2);
  \fill[axisblue] (\scXshr,\scYc2) circle (0.07);
}
