% The quarter-wave monopole as it actually ships, drawn the way you
% see it on a board: plan view, looking straight down. Solder mask,
% ground pour, the radio, the antenna trace on a keep-out with the
% copper cut away beneath it. Above the board, on the same horizontal
% scale, the current standing wave the trace carries - maximum at the
% feed, zero at the tip, which is what makes l = lambda/4 the natural
% length. About 31 mm at 2.4 GHz, which is why the antenna end of a
% Bluetooth board is the size it is.

\begin{tikzpicture}[thick]

  \def\xfeed{4.6}
  \def\xtip{9.2}

  % ---- the current standing wave, on the trace's own scale ----
  \begin{scope}[shift={(0,2.5)}]
    \draw[gray!60, ->] (\xfeed-0.4,0) -- (\xtip+0.7,0);
    \node[gray!60!black, anchor=west, scale=0.85] at (\xtip+0.75,0) {$z$};
    \draw[blue, very thick]
      (\xfeed,1.35) .. controls (\xfeed+2.2,1.3) and (\xtip-1.3,0.85) .. (\xtip,0);
    \node[blue, anchor=south west, scale=0.9] at (\xfeed+1.5,1.1) {$I(z)$};
    \draw[gray!60, dashed] (\xfeed,0) -- (\xfeed,1.35);
    \node[anchor=north east, scale=0.8] at (\xfeed-0.05,-0.05) {feed: $I$ max};
    \node[anchor=north west, scale=0.8] at (\xtip+0.05,-0.05) {tip: $I=0$};
  \end{scope}

  % ---- the board, seen from above ----
  \fill[armygreen!18, rounded corners=3pt] (0,-2.6) rectangle (10.2,1.0);
  \draw[armygreen!45, rounded corners=3pt] (0,-2.6) rectangle (10.2,1.0);

  % ground pour: copper everywhere except the antenna keep-out
  \fill[orange!35] (0.3,-2.3) rectangle (\xfeed-0.3,0.7);
  \node[orange!55!black, anchor=center, scale=0.85] at (2.1,-1.4) {ground pour};

  % the keep-out: no copper under or beside the trace
  \draw[gray!70, dashed] (\xfeed-0.3,-2.3) rectangle (10.0,0.7);
  \node[gray!60!black, anchor=north, scale=0.8] at (7.4,0.62)
    {keep-out: no copper here};

  % the radio, sitting on the pour edge
  \fill[black!80, rounded corners=1pt] (3.3,-0.85) rectangle (\xfeed,0.15);
  \node[white, anchor=center, scale=0.75] at (3.95,-0.35) {radio};

  % the antenna trace: a copper strip from the feed to the tip
  \fill[orange!85] (\xfeed,-0.55) rectangle (\xtip,-0.15);
  \fill[orange!85] (\xfeed-0.15,-0.5) rectangle (\xfeed,-0.2);

  % the length that matters
  \draw[<->] (\xfeed,-1.35) -- (\xtip,-1.35);
  \node[anchor=north, scale=0.9] at ({(\xfeed+\xtip)/2},-1.45)
    {$l = \lambda/4 \approx 31$ mm at 2.4 GHz};

  % the pour is the other half of the antenna
  \draw[gray, thin] (2.1,0.7) -- (2.1,1.35);
  \node[anchor=south, scale=0.85, align=center] at (2.1,1.35)
    {the pour is the other\\half of the antenna};

\end{tikzpicture}
\end{document}
