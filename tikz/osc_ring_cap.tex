% Digital frequency control of a ring oscillator stage: binary weighted
% capacitors C, 2C, 4C switched onto the inverter output by the digital
% control word D0, D1, D2.

\begin{circuitikz}[american, thick, transform shape, circuit ee IEC]

  \input{tikz/ckt_lib.tex}

  \def\yr{0}      % the output rail
  \def\xa{1.6}    % branch columns
  \def\xb{4.2}
  \def\xc{6.8}

  % one stage of the ring
  \draw (-3.4,\yr) node[anchor=east] {from ring} to [short,o-] (-2.6,\yr);
  \draw (-2.35,\yr) \cicInv{inv};
  \draw (inv_out) -- (8.2,\yr);
  \draw (8.2,\yr) to [short,-o] ++(0.5,0) node[anchor=west] {to ring};

  % binary weighted branches: NMOS switch over capacitor
  \draw (\xa,{\yr-1.9}) \lvnmos{MS0}{$D_0$};
  \draw (\xa,\yr) to [short,*-] (\xa,{\yr-0.3});
  \draw (\xa,{\yr-1.9}) to [C, l=$C$] (\xa,{\yr-3.3});
  \draw (\xa,{\yr-3.3}) \vground;

  \draw (\xb,{\yr-1.9}) \lvnmos{MS1}{$D_1$};
  \draw (\xb,\yr) to [short,*-] (\xb,{\yr-0.3});
  \draw (\xb,{\yr-1.9}) to [C, l=$2C$] (\xb,{\yr-3.3});
  \draw (\xb,{\yr-3.3}) \vground;

  \draw (\xc,{\yr-1.9}) \lvnmos{MS2}{$D_2$};
  \draw (\xc,\yr) to [short,*-] (\xc,{\yr-0.3});
  \draw (\xc,{\yr-1.9}) to [C, l=$4C$] (\xc,{\yr-3.3});
  \draw (\xc,{\yr-3.3}) \vground;

\end{circuitikz}
\end{document}
