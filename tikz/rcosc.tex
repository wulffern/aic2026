% Relaxation (RC) oscillator: a mirror forces I into a resistor to make
% the reference V1 = R*I, and into a capacitor so V2 ramps. When V2
% crosses V1 the comparator fires, a delayed pulse resets the capacitor
% through the NMOS switch, and the flip flop divides the pulse train by
% two to a square wave output f_o.

\begin{circuitikz}[american, thick, transform shape, circuit ee IEC]

  \input{tikz/ckt_lib.tex}

  \def\ydr{4.2}   % mirror drain row
  \def\ytop{5.8}  % supply row
  \def\xr{2.6}    % the resistor branch
  \def\xc{5.2}    % the capacitor branch
  \def\yva{3.0}   % the V1 wire
  \def\yvb{2.2}   % the V2 wire

  % ---- PMOS mirror: diode device and two outputs ----
  \draw (0,\ytop) \vsupply;
  \draw (0,\ydr) \lvmpmos{MD}{};
  \coordinate (gmd) at (MD.G);
  \fill (gmd) circle (0.075);
  \draw (gmd) |- (0,\ydr);
  \fill (0,\ydr) circle (0.075);
  \draw (0,\ydr) to [I, l_=$I$] (0,1.0);
  \draw (0,1.0) \vground;

  \draw (\xr,\ytop) \vsupply;
  \draw (\xr,\ydr) \lvpmos{M2}{};
  \draw (\xc,\ytop) \vsupply;
  \draw (\xc,\ydr) \lvpmos{M3}{};
  \coordinate (g2) at (M2.G);
  \coordinate (g3) at (M3.G);
  \draw (gmd) -- (g2);
  \draw (g2) -- (g3);
  \fill (g2) circle (0.075);

  % ---- V1 = R*I ----
  \draw (\xr,\ydr) -- (\xr,\yva);
  \fill (\xr,\yva) circle (0.075);
  \draw (\xr,\yva) to [R, l=$R$] (\xr,0.8);
  \draw (\xr,0.8) \vground;
  \node[anchor=south east] at (\xr,{\yva+0.1}) {$V_1$};

  % ---- V2 ramps on C, reset switch beside it ----
  \draw (\xc,\ydr) -- (\xc,\yvb);
  \fill (\xc,\yvb) circle (0.075);
  \draw (\xc,\yvb) to [C, l_=$C$] (\xc,0.8);
  \draw (\xc,0.8) \vground;
  \node[anchor=south east] at (\xc,{\yvb+0.1}) {$V_2$};

  \draw (6.6,0.4) \lvmnmos{MSW}{};
  \draw (6.6,0.4) \vground;
  \draw (6.6,2.0) -- (6.6,\yvb) -- (\xc,\yvb);

  % ---- comparator ----
  \draw (8.0,3.4) -- (8.0,1.6) -- (9.8,2.5) -- cycle;
  \node[anchor=west, scale=0.9] at (8.05,3.0) {$-$};
  \node[anchor=west, scale=0.9] at (8.05,2.0) {$+$};
  \draw (\xr,\yva) -- (7.4,\yva) -- (7.4,3.0) -- (8.0,3.0);
  \draw (6.6,\yvb) -- (7.6,\yvb) -- (7.6,2.0) -- (8.0,2.0);
  \fill (6.6,\yvb) circle (0.075);
  \draw (9.8,2.5) -- (10.8,2.5);
  \fill (10.4,2.5) circle (0.075);

  % ---- delayed reset pulse back to the switch ----
  \node[buffer port, rotate=180] (b1) at (9.6,-0.4) {};
  \node[buffer port, rotate=180] (b2) at (8.2,-0.4) {};
  \draw (10.4,2.5) -- (10.4,-0.4) -- (b1.in);
  \draw (b1.out) -- (b2.in);
  \coordinate (gsw) at (MSW.G);
  \draw (b2.out) -- (7.3,-0.4) -- (7.3,1.2) -- (gsw);

  % ---- divide by two flip flop ----
  \draw (10.8,1.6) rectangle (13.2,4.4);
  \node[anchor=west] at (10.95,3.8) {D};
  \node[anchor=east] at (13.05,3.8) {Q};
  \node[anchor=east] at (13.05,2.2) {$\overline{\mbox{Q}}$};
  \draw (10.8,2.3) -- (11.15,2.5) -- (10.8,2.7);

  % Qbar back to D
  \draw (13.2,2.2) -- (13.9,2.2) -- (13.9,5.0) -- (10.2,5.0)
        -- (10.2,3.8) -- (10.8,3.8);

  % output
  \draw (13.2,3.8) to [short,-o] (14.3,3.8);
  \node[anchor=west] at (14.35,3.8) {$f_o$};

\end{circuitikz}
