% LC oscillator: a PMOS mirror biases the tank at the center tap of the
% inductor, the tank capacitors are grounded in the middle, two MOS
% varactors tune the frequency with Vcnt, and a cross coupled NMOS pair
% provides the negative resistance. Wires that cross without a dot are
% not connected.

\begin{circuitikz}[american, thick, transform shape, circuit ee IEC]

  \input{tikz/ckt_lib.tex}

  \def\xa{0}    % left tank node
  \def\xb{5}    % right tank node
  \def\xc{2.5}  % center
  \def\ytk{6.4} % inductor row
  \def\ycp{5.2} % capacitor row
  \def\yvt{4.25}% varactor top
  \def\yvb{3.55}% varactor bottom

  % ---- bias: PMOS mirror on top ----
  \draw (-1.7,7.0) to [Tpmos, n=PM1, mirror] (-1.7,8.6);
  \draw (-1.7,8.6) \vsupply;
  \draw (\xc,7.0) to [Tpmos, n=PM2] (\xc,8.6);
  \draw (\xc,8.6) \vsupply;
  \draw (PM1.G) -- (PM2.G);
  \fill (PM1.G) circle (0.07);
  \draw (PM1.G) |- (-1.7,7.0);
  \fill (-1.7,7.0) circle (0.07);
  \draw (-1.7,7.0) -- (-1.7,6.2) to [I] (-1.7,4.6) -- (-1.7,4.3) \vground;
  \draw (\xc,7.0) -- (\xc,\ytk);
  \fill (\xc,\ytk) circle (0.07);

  % ---- the tank: two inductors from the center tap ----
  \draw (\xa,\ytk) to [L] (\xc,\ytk);
  \draw (\xc,\ytk) to [L, l=$L$] (\xb,\ytk);

  % side rails
  \draw (\xa,\ytk) -- (\xa,2.6);
  \draw (\xb,\ytk) -- (\xb,2.6);

  % ---- tank capacitors, grounded in the middle ----
  \draw (\xa,\ycp) to [C, l=$C$] (\xc,\ycp);
  \draw (\xc,\ycp) to [C, l=$C$] (\xb,\ycp);
  \fill (\xa,\ycp) circle (0.07);
  \fill (\xb,\ycp) circle (0.07);
  \draw (\xc,\ycp) -- ++(0,-0.3) \vground;

  % ---- MOS varactors, tuned by Vcnt ----
  % left device, gate towards the left rail, source and drain tied
  \draw (\xa,3.9) -- (1.3,3.9);
  \fill (\xa,3.9) circle (0.07);
  \draw (1.3,\yvb) -- (1.3,\yvt);
  \draw (1.45,\yvb) -- (1.45,\yvt);
  \draw (1.45,\yvt) -- (1.8,\yvt) -- (1.8,\yvb) -- (1.45,\yvb);
  \draw (1.8,3.9) -- (\xc,3.9);
  % right device, gate towards the right rail, source and drain tied
  \draw (\xb,3.9) -- (3.7,3.9);
  \fill (\xb,3.9) circle (0.07);
  \draw (3.7,\yvb) -- (3.7,\yvt);
  \draw (3.55,\yvb) -- (3.55,\yvt);
  \draw (3.55,\yvt) -- (3.2,\yvt) -- (3.2,\yvb) -- (3.55,\yvb);
  \draw (3.2,3.9) -- (\xc,3.9);
  % common node up to the tuning port
  \fill (\xc,3.9) circle (0.07);
  \draw (\xc,3.9) -- (\xc,4.55) -- (-0.7,4.55) to [short,-o] (-0.9,4.55);
  \node[anchor=north] at (-0.85,4.35) {$V_{cnt}$};

  % ---- cross coupled NMOS pair ----
  \draw (\xa,1.0) to [Tnmos, n=MN1, mirror] (\xa,2.6);
  \draw (\xa,1.0) -- ++(0,-0.3) \vground;
  \draw (\xb,1.0) to [Tnmos, n=MN2] (\xb,2.6);
  \draw (\xb,1.0) -- ++(0,-0.3) \vground;

  % each gate to the OTHER drain
  \draw (MN1.G) -- (\xb,2.9);
  \fill (\xb,2.9) circle (0.07);
  \draw (MN2.G) -- (\xa,2.9);
  \fill (\xa,2.9) circle (0.07);

\end{circuitikz}
\end{document}
