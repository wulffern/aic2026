% Mechanical stress components on the channel, drawn with one consistent
% oblique projection: F_y vertical, F_x along the current direction,
% F_z along the width (the depth axis).

\definecolor{echarge}{RGB}{85,155,235}
\definecolor{hcharge}{RGB}{160,25,35}
\definecolor{sdgreen}{RGB}{70,130,60}
\definecolor{oxteal}{RGB}{140,195,195}

\begin{tikzpicture}[scale=0.9]
  % depth vector for the oblique projection
  \def\dx{1.3}
  \def\dy{0.85}

  % channel floor
  \fill[gray!12] (0.2,0) -- (8.2,0) -- (8.2+\dx,\dy) -- (0.2+\dx,\dy) -- cycle;
  \draw[gray!60] (0.2,0) -- (8.2,0) -- (8.2+\dx,\dy) -- (0.2+\dx,\dy) -- cycle;

  % source block (front, top, side faces)
  \fill[sdgreen!75] (0.2,0) rectangle (1.7,0.9);
  \fill[sdgreen!55] (0.2,0.9) -- (1.7,0.9) -- (1.7+\dx,0.9+\dy) -- (0.2+\dx,0.9+\dy) -- cycle;
  \fill[sdgreen!40] (1.7,0) -- (1.7,0.9) -- (1.7+\dx,0.9+\dy) -- (1.7+\dx,\dy) -- cycle;
  \node[white, scale=1.3] at (0.95,0.45) {S};

  % drain block
  \fill[sdgreen!75] (6.7,0) rectangle (8.2,0.9);
  \fill[sdgreen!55] (6.7,0.9) -- (8.2,0.9) -- (8.2+\dx,0.9+\dy) -- (6.7+\dx,0.9+\dy) -- cycle;
  \fill[sdgreen!40] (8.2,0) -- (8.2,0.9) -- (8.2+\dx,0.9+\dy) -- (8.2+\dx,\dy) -- cycle;
  \node[white, scale=1.3] at (7.45,0.45) {D};

  % thin gate oxide between the channel and the gate slab
  \fill[oxteal!60] (2.3,1.75) rectangle (6.1,1.95);
  \fill[oxteal!45] (2.3,1.95) -- (6.1,1.95) -- (6.1+\dx,1.95+\dy) -- (2.3+\dx,1.95+\dy) -- cycle;
  \fill[oxteal!35] (6.1,1.75) -- (6.1,1.95) -- (6.1+\dx,1.95+\dy) -- (6.1+\dx,1.75+\dy) -- cycle;

  % gate slab floating above the channel
  \fill[hcharge!65] (2.3,2.2) rectangle (6.1,3.1);
  \fill[hcharge!45] (2.3,3.1) -- (6.1,3.1) -- (6.1+\dx,3.1+\dy) -- (2.3+\dx,3.1+\dy) -- cycle;
  \fill[hcharge!35] (6.1,2.2) -- (6.1,3.1) -- (6.1+\dx,3.1+\dy) -- (6.1+\dx,2.2+\dy) -- cycle;
  \node[black!60, scale=1.3] at (4.2,2.65) {G};

  % the three force components, well separated
  \draw[echarge, line width=1.8pt, ->] (4.2,4.6) -- (4.2,3.65);
  \node[echarge, anchor=west, scale=1.1] at (4.4,4.25) {$F_y$};

  \draw[echarge, line width=1.8pt, ->] (2.5,0.45) -- (5.9,0.45);
  \node[echarge, anchor=south, scale=1.1] at (5.3,0.55) {$F_x$};

  \draw[echarge, line width=1.8pt, ->] (3.1,1.05) -- (3.1+\dx,1.05+\dy);
  \node[echarge, anchor=east, scale=1.1] at (3.6,1.75) {$F_z$};

\end{tikzpicture}
\end{document}
