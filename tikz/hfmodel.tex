% High frequency small signal model. The low frequency core (g_m, g_s,
% r_ds) plus the four capacitances: C_gs gate-source, C_gd gate-drain,
% C_db and C_sb from drain and source to the (AC grounded) bulk.

\begin{circuitikz}[american, thick, transform shape]

  \input{tikz/ckt_lib.tex}

  \def\ytop{2.8}   % drain rail
  \def\ymid{1.4}   % gate height
  \def\xg{1.2}     % gate column (C_gs)
  \def\xgm{3.0}
  \def\xgs{5.2}
  \def\xrds{6.6}
  \def\xterm{7.8}

  % Gate terminal and column
  \draw (0.09,\ymid) circle (0.09);
  \node[anchor=east] at (-0.1,\ymid) {$v_g$};
  \draw (0.18,\ymid) -- (\xg,\ymid);
  \fill (\xg,\ymid) circle (0.075);

  % C_gs down to the source rail
  \draw (\xg,\ymid) to [C, l=$C_{gs}$] (\xg,0);

  % C_gd along the top to the drain rail
  \draw (\xg,\ymid) -- (\xg,\ytop);
  \draw (\xg,\ytop) to [C, l=$C_{gd}$] (\xgm,\ytop);

  % Low frequency core between the rails
  \draw (\xgm,\ytop) to [cI, l_=$g_m v_{gs}$] (\xgm,0);
  \draw (\xgs,0) to [cI, l=$g_s v_{sb}$] (\xgs,\ytop);
  \draw (\xrds,\ytop) to [R, l=$r_{ds}$] (\xrds,0);

  % Drain rail, C_db up to the AC grounded bulk
  \draw (\xgm,\ytop) -- (\xterm,\ytop);
  \fill (\xgm,\ytop) circle (0.075);
  \fill (\xgs,\ytop) circle (0.075);
  \fill (\xrds,\ytop) circle (0.075);
  \draw (4.1,\ytop) to [C, l=$C_{db}$] (4.1,{\ytop+1.2});
  \fill (4.1,\ytop) circle (0.075);
  % ground pointing up: the bars shrink away from the wire
  \draw (3.9,{\ytop+1.2}) -- (4.3,{\ytop+1.2});
  \draw (3.975,{\ytop+1.3}) -- (4.225,{\ytop+1.3});
  \draw (4.05,{\ytop+1.4}) -- (4.15,{\ytop+1.4});
  \draw (\xterm+0.09,\ytop) circle (0.09);
  \node[anchor=west] at ({\xterm+0.3},\ytop) {$v_d$};
  \draw[->] ({\xterm-0.4},{\ytop+0.3}) -- ({\xterm-1.2},{\ytop+0.3});
  \node[anchor=south] at ({\xterm-0.8},{\ytop+0.35}) {$i_d$};

  % Source rail, C_sb down to the AC grounded bulk
  \draw (\xg,0) -- (\xterm,0);
  \fill (\xg,0) circle (0.075);
  \fill (\xgm,0) circle (0.075);
  \fill (\xgs,0) circle (0.075);
  \fill (\xrds,0) circle (0.075);
  \draw (2.1,0) to [C, l_=$C_{sb}$] (2.1,-1.2);
  \fill (2.1,0) circle (0.075);
  \draw (2.1,-1.2) \vground;
  \draw (\xterm+0.09,0) circle (0.09);
  \node[anchor=west] at ({\xterm+0.3},0) {$v_s$};
  \draw[->] ({\xterm-0.8},-0.25) -- ({\xterm-0.8},-0.85);
  \node[anchor=east] at ({\xterm-0.9},-0.55) {$i_s$};

\end{circuitikz}
\end{document}
