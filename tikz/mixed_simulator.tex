% The mixed-signal simulator, redrawn from the hand sketch: a digital
% and an analog simulator side by side, exchanging values only through
% connect modules - DACs on the way into the analog world, ADCs on the
% way back - with a shared event and timestep control under both. The
% point of the drawing is that nothing else crosses the boundary.

\begin{tikzpicture}[thick]

  \draw (0,0) rectangle (3.4,7.0);
  \node[align=center] at (1.7,5.9) {Digital\\simulator};
  \draw (8.6,0) rectangle (12.0,7.0);
  \node[align=center] at (10.3,5.9) {Analog\\simulator};

  % Connect modules: a pentagon per crossing signal
  \newcommand{\simDac}[2]{%
    \draw (#1,#2) -- ++(1.4,0) -- ++(0.55,-0.45) -- ++(-0.55,-0.45) -- ++(-1.4,0) -- cycle;
    \node at ({#1+0.85},{#2-0.45}) {DAC};
  }
  \newcommand{\simAdc}[2]{%
    \draw (#1,#2) -- ++(-1.4,0) -- ++(-0.55,-0.45) -- ++(0.55,-0.45) -- ++(1.4,0) -- cycle;
    \node at ({#1-0.85},{#2-0.45}) {ADC};
  }
  \simDac{5.0}{6.2}
  \simDac{5.0}{4.6}
  \simAdc{7.0}{2.9}
  \simAdc{7.0}{1.3}
  \draw[->] (3.4,5.75) -- (5.0,5.75);
  \draw[->] (6.95,5.75) -- (8.6,5.75);
  \draw[->] (3.4,4.15) -- (5.0,4.15);
  \draw[->] (6.95,4.15) -- (8.6,4.15);
  \draw[->] (8.6,2.45) -- (7.0,2.45);
  \draw[->] (5.05,2.45) -- (3.4,2.45);
  \draw[->] (8.6,0.85) -- (7.0,0.85);
  \draw[->] (5.05,0.85) -- (3.4,0.85);
  \fill (6.0,3.62) circle (0.035);
  \fill (6.0,3.42) circle (0.035);
  \fill (6.0,3.22) circle (0.035);

  \draw (0,-2.4) rectangle (12.0,-1.0);
  \node at (6.0,-1.7) {Event and timestep control};
  \draw[->] (1.4,-1.0) -- (1.4,0);
  \draw[<-] (2.0,-1.0) -- (2.0,0);
  \draw[->] (10.0,-1.0) -- (10.0,0);
  \draw[<-] (10.6,-1.0) -- (10.6,0);

\end{tikzpicture}
