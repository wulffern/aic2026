% Source follower, small signal. The drain is at AC ground (top rail), the
% source is the output (bottom rail). Three elements hang between them:
% g_m v_gs, the bulk source g_s v_s, and r_ds. The input v_i drives the
% gate, and v_gs = v_i - v_o.

\begin{circuitikz}[american, thick, transform shape, circuit ee IEC]

  \input{tikz/ckt_lib.tex}

  \def\ytop{2.6}
  \def\xgm{3.6}
  \def\xgs{5.6}
  \def\xrds{7.6}
  \def\xterm{8.8}

  % input source at the far left
  \draw (0,-1.3) \vground;
  \draw (0,\ytop) to [V, l_=$v_i$] (0,-1.3);
  \draw (0,\ytop) -- (1.3,\ytop);

  % floating gate marker with v_gs annotation down to the output rail
  \draw (1.39,\ytop) circle (0.09);
  \node[anchor=south] at (1.2,{\ytop+0.1}) {G};
  \node[anchor=north] at (1.4,{\ytop-0.25}) {$+$};
  \node[anchor=west, color=blue] at (0.9,1.3) {$v_{gs}$};
  \node[anchor=south] at (1.4,0.25) {$-$};

  % elements between drain rail (AC ground) and source rail (output)
  \draw (\xgm,\ytop) to [cI, l_=$g_m v_{gs}$] (\xgm,0);
  \draw (\xgs,0) to [cI, l=$g_s v_{s}$] (\xgs,\ytop);
  \draw (\xrds,0) -- (\xrds,0.5) \vresistor{$r_{ds}$} -- (\xrds,2.6);

  % drain rail: AC ground
  \draw (\xgm,\ytop) -- (\xterm,\ytop);
  \fill (\xgs,\ytop) circle (0.075);
  \fill (\xrds,\ytop) circle (0.075);
  \draw (\xterm,\ytop) \vground;
  \node[anchor=south] at (\xterm,{\ytop+0.1}) {D};

  % source rail: the output
  \draw (\xgm,0) -- (\xterm,0);
  \fill (\xgs,0) circle (0.075);
  \fill (\xrds,0) circle (0.075);
  \draw (\xterm,0) \portOut{$v_o$};
  \node[anchor=north] at (\xterm,-0.3) {S};

\end{circuitikz}
\end{document}
