% Folded cascode OTA. NMOS input pair in the middle, and the signal
% current folds outwards into the cascoded output branches: PMOS current
% sources on top, PMOS cascodes, NMOS cascoded mirror at the bottom.

\begin{circuitikz}[american, thick, transform shape, circuit ee IEC]

  \input{tikz/ckt_lib.tex}

  \def\xa{-2.6}     % left fold branch
  \def\xl{0}
  \def\xr{3.2}
  \def\xm{1.6}
  \def\xb{5.8}      % right fold branch
  \def\ysrc{2.0}    % pair source level
  \def\ymir{0}      % mirror row
  \def\ycas{1.6}    % NMOS cascode row (\ymir + \grid)
  \def\yout{3.2}    % output level (\ycas + \grid)
  \def\ypcas{3.2}   % PMOS cascode row bottom
  \def\yfold{4.8}   % fold node (\ypcas + \grid)
  \def\ytop{6.4}    % supply (\yfold + \grid)

  % ---- input pair with tail ----
  \draw (\xm,-0.2) \vground;
  \draw (\xm,-0.2) \lvnmos{MT}{$V_{B4}$};
  \draw (\xl,\ysrc) -- (\xl,1.7) -- (\xr,1.7) -- (\xr,\ysrc);
  \draw (\xm,1.4) -- (\xm,1.7);
  \fill (\xm,1.7) circle (0.075);

  \draw (\xl,\ysrc) \lvnmos{M1}{$v_p$};
  \draw (\xr,\ysrc) \lvmnmos{M2}{$v_n$};

  % fold wires from the pair drains out to the fold nodes
  \draw (\xl,{\ysrc+\grid}) -- (\xl,\yfold) -- (\xa,\yfold);
  \draw (\xr,{\ysrc+\grid}) -- (\xr,\yfold) -- (\xb,\yfold);
  \fill (\xa,\yfold) circle (0.075);
  \fill (\xb,\yfold) circle (0.075);

  % ---- top PMOS current sources ----
  \draw (\xa,\yfold) \lvpmos{M9}{$V_{B1}$};
  \draw (\xb,\yfold) \lvmpmos{M10}{$V_{B1}$};
  \draw (\xa,\ytop) \vsupply;
  \draw (\xb,\ytop) \vsupply;

  % ---- PMOS cascodes ----
  \draw (\xa,\ypcas) \lvpmos{M7}{$V_{B2}$};
  \draw (\xb,\ypcas) \lvmpmos{M8}{$V_{B2}$};

  % ---- NMOS cascodes ----
  \draw (\xa,\ycas) \lvnmos{M5}{$V_{B3}$};
  \draw (\xb,\ycas) \lvmnmos{M6}{$V_{B3}$};

  % ---- NMOS mirror at the bottom ----
  \draw (\xa,\ymir) \vground;
  \draw (\xa,\ymir) \lvnmos{M3}{};
  \draw (\xb,\ymir) \vground;
  \draw (\xb,\ymir) \lvmnmos{M4}{};
  \coordinate (g3) at (M3.G);
  \coordinate (g4) at (M4.G);
  % diode connect the left branch at the cascode drain, routed wide of
  % the bias labels
  \fill (\xa,\ypcas) circle (0.075);
  \fill (g3) circle (0.075);
  \draw (g3) -- ++(-1.3,0) coordinate (g3a);
  \draw (g3a) -- (g3a |- 0,3.6) coordinate (g3b);
  \draw (g3b) -- (-2.9,3.6) -- (-2.9,\ypcas) -- (\xa,\ypcas);
  % mirror gate wire routed underneath the ground row
  \draw (g3) -- ++(0,-1.5) coordinate (g3c);
  \draw (g4) -- ++(0,-1.5) coordinate (g4c);
  \draw (g3c) -- (g4c);

  % output
  \draw (\xb,\yout) \portOut{$v_o$};
  \fill (\xb,\yout) circle (0.075);

  % device names
  \node[anchor=west, scale=0.8] at (0.35,2.8) {M1};
  \node[anchor=east, scale=0.8] at (2.85,2.8) {M2};
  \node[anchor=west, scale=0.8] at (-2.25,0.8) {M3};
  \node[anchor=east, scale=0.8] at (5.45,0.8) {M4};
  \node[anchor=west, scale=0.8] at (-2.25,2.4) {M5};
  \node[anchor=east, scale=0.8] at (5.45,2.4) {M6};
  \node[anchor=west, scale=0.8] at (-2.25,4.0) {M7};
  \node[anchor=east, scale=0.8] at (5.45,4.0) {M8};
  \node[anchor=west, scale=0.8] at (-2.25,5.6) {M9};
  \node[anchor=east, scale=0.8] at (5.45,5.6) {M10};
  \node[anchor=west, scale=0.8] at (1.95,0.3) {MT};

\end{circuitikz}
