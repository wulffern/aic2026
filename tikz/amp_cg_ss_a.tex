% Common gate small signal gain. The input v_i drives the source, the
% output hangs open at the drain (i_o = 0), and v_gs = -v_i.

\begin{circuitikz}[american, thick, transform shape, circuit ee IEC]

  \input{tikz/ckt_lib.tex}

  \def\ytop{2.6}
  \def\xgm{4.2}
  \def\xrds{6.0}
  \def\xterm{7.2}

  % grounded gate marker with v_gs annotation
  \draw (1.1,{\ytop+0.9}) \vground;
  \draw (1.1,{\ytop+0.9}) -- (1.4,{\ytop+0.9}) -- (1.4,{\ytop+0.09});
  \draw (1.4,\ytop) circle (0.09);
  \node[anchor=east] at (1.2,\ytop) {G};
  \node[anchor=north] at (1.4,{\ytop-0.25}) {$+$};
  \node[anchor=west, color=blue] at (1.0,1.3) {$v_{gs}=-v_i$};
  \node[anchor=south] at (1.4,0.25) {$-$};

  % elements between drain rail (output, top) and source rail (input)
  \draw (\xgm,\ytop) to [cI, l_=$g_m v_{gs}$] (\xgm,0);
  \draw (\xrds,\ytop) to [R, l_=$r_{ds}$] (\xrds,0);

  % drain rail: open output
  \draw (\xgm,\ytop) -- (\xterm,\ytop);
  \fill (\xrds,\ytop) circle (0.075);
  \node[anchor=south] at (\xgm,{\ytop+0.1}) {D};
  \draw (\xterm,\ytop) \portOut{$v_o$};

  % source rail driven by the input
  \draw (\xgm,0) -- (\xterm,0);
  \fill (\xrds,0) circle (0.075);
  \node[anchor=north] at (\xgm,-0.15) {S};
  \draw (\xterm,0) to [short,-] ({\xterm+1.6},0);
  \draw ({\xterm+1.6},0) to [V, l_=$v_i$] ({\xterm+1.6},-1.6);
  \draw ({\xterm+1.6},-1.6) \vground;

\end{circuitikz}
