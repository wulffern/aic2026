% pinched-off region gain enough energy to create electron-hole pairs by
% impact ionization; some are injected into the oxide.

\begin{tikzpicture}[scale=0.85]
  \input{tikz/mos_lib.tex}
  \drainzoomframe

  % field from the gate down through the oxide
  \foreach \x in {1.3,2.0,2.7}{

  }
  % field at the drain junction
  \foreach \x in {7.4,8.2,9.0}{
    \draw[fieldy, thick, -latex] (\x,2.05) -- (\x,1.65);
  }

  % one primary electron, accelerated across the gap
  \node[echarge, anchor=center, scale=1.15] at (3.6,4.35) {$\ominus$};
  \draw[echarge, line width=1.3pt, -latex] (3.9,4.33) -- (5.6,4.08);

  % impact ionization: a small burst at the impact point
  \foreach \a in {20,80,150,210,300}{
    \draw[black!60, thin] (5.85,4.0) -- ++(\a:0.18);
  }

  % products: an electron kicked up towards the oxide, a hole driven
  % into the bulk, and the primary carrying on to the drain
  \node[echarge, anchor=center, scale=1.05] at (6.15,4.35) {$\ominus$};
  \draw[echarge, line width=1.2pt, -latex] (6.25,4.5) -- (6.5,5.05);
