% Diffusion resistors in cross section. The substrate is p- in every
% modern process, so the n-diffusion resistor sits straight in it,
% while a p-diffusion resistor needs an n-well around it - and that
% well has to be tied to a potential of its own.

\begin{tikzpicture}[thick]

  \definecolor{nwellc}{RGB}{190,215,255}

  % ---- the shared p- substrate ----
  \draw[fill=black!4] (0,0) rectangle (13.4,2.6);
  \node[anchor=south west, color=black!60] at (0.2,0.15) {p$-$ substrate};

  % squiggle showing the distributed resistance of a strip
  \newcommand{\squiggle}[2]{%
    \draw[densely dotted, black!60]
      ({#1+0.5},{#2+0.35}) -- ({#1+0.9},{#2+0.35}) -- ({#1+1.1},{#2+0.55})
      -- ({#1+1.5},{#2+0.15}) -- ({#1+1.9},{#2+0.55}) -- ({#1+2.3},{#2+0.15})
      -- ({#1+2.7},{#2+0.55}) -- ({#1+3.1},{#2+0.15}) -- ({#1+3.3},{#2+0.35})
      -- ({#1+3.8},{#2+0.35});}

  % contact + metal terminal at (x, top of silicon), labelled #3
  \newcommand{\term}[3]{%
    \fill[cut] ({#1-0.17},#2) rectangle ({#1+0.17},{#2+0.35});
    \fill[mOne, rounded corners=1.5pt]
      ({#1-1.0},{#2+0.35}) rectangle ({#1+1.0},{#2+0.8});
    \node[anchor=south] at (#1,{#2+0.85}) {#3};}

  % ---- left: n-diffusion straight in the substrate ----
  \fill[active] (0.9,1.9) rectangle (5.2,2.6);
  \draw (0.9,1.9) rectangle (5.2,2.6);
  \node[anchor=north, color=black!60] at (3.05,1.8) {n$+$ diffusion};
  \squiggle{0.9}{1.9}
  \term{1.4}{2.6}{A}
  \term{4.7}{2.6}{B}

  % ---- right: p-diffusion inside an n-well ----
  \fill[nwellc] (7.2,0.9) rectangle (12.9,2.6);
  \draw (7.2,0.9) rectangle (12.9,2.6);
  \node[anchor=south west, color=black!60] at (7.35,1.0) {n-well};

  \fill[poly] (7.9,1.9) rectangle (12.2,2.6);
  \draw (7.9,1.9) rectangle (12.2,2.6);
  \node[anchor=north, color=black!60] at (10.05,1.75) {p$+$ diffusion};
  \squiggle{7.9}{1.9}
  \term{8.4}{2.6}{A};
  \term{11.7}{2.6}{B};

  % the well tie: the well is a terminal too, and must be biased
  \fill[active] (12.4,2.2) rectangle (12.85,2.6);
  \draw (12.4,2.2) rectangle (12.85,2.6);
  \fill[cut] (12.5,2.6) rectangle (12.75,2.95);
  \fill[mOne, rounded corners=1.5pt] (12.3,2.95) rectangle (13.3,3.4);
  \node[anchor=south] at (12.8,3.45) {well};

\end{tikzpicture}
\end{document}
