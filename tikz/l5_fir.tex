% Three tap FIR: x[n] and two delayed copies, each weighted 1/3, summed.
% Matches the transfer function on the same slide, H(z) = (1/3) sum z^-i.

\definecolor{axisblue}{rgb}{0.16,0.16,0.55}

\begin{circuitikz}[american, transform shape]

  \def\ysig{0}
  \def\ysum{-2.1}

  % ---- Signal path: two delays in series ----
  \draw[axisblue, thick] (-1.9,\ysig) -- (0.4,\ysig);
  \node[axisblue, anchor=east] at (-2.0,\ysig) {$x[n]$};

  \draw[axisblue, thick] (0.4,\ysig) rectangle ++(1.5,0.95);
  \node[axisblue] at (1.15,\ysig+0.48) {$z^{-1}$};
  \draw[axisblue, thick, ->] (1.9,\ysig) -- (2.8,\ysig);

  \draw[axisblue, thick] (2.8,\ysig) rectangle ++(1.5,0.95);
  \node[axisblue] at (3.55,\ysig+0.48) {$z^{-1}$};
  \draw[axisblue, thick] (4.3,\ysig) -- (5.0,\ysig);

  % The delay blocks sit on the signal line, so the line enters and leaves at
  % their lower left and lower right corners.
  \draw[axisblue, thick, ->] (0.0,\ysig) -- (0.4,\ysig);

  % ---- Summing node ----
  \draw[axisblue, thick] (6.0,\ysum) circle (0.3);
  \node[axisblue] at (6.0,\ysum) {$+$};
  \draw[axisblue, thick, ->] (6.3,\ysum) -- (7.2,\ysum);
  \node[axisblue, anchor=west] at (7.3,\ysum) {$y[n]$};

  % ---- Three taps, each a third ----
  % Straight through from the input.
  \draw[axisblue, thick] (-1.0,\ysig) -- (-1.0,\ysum-1.1) -- (5.2,\ysum-1.1);
  \draw[axisblue, thick, ->] (5.2,\ysum-1.1) -- (5.78,\ysum-0.22);
  \fill[axisblue] (-1.0,\ysig) circle (0.07);
  \node[axisblue, anchor=north] at (2.0,\ysum-1.2) {$\frac{1}{3}$};

  % After one delay.
  \draw[axisblue, thick] (2.35,\ysig) -- (2.35,\ysum-0.55) -- (5.3,\ysum-0.55);
  \draw[axisblue, thick, ->] (5.3,\ysum-0.55) -- (5.78,\ysum-0.15);
  \fill[axisblue] (2.35,\ysig) circle (0.07);
  \node[axisblue, anchor=north] at (3.9,\ysum-0.62) {$\frac{1}{3}$};

  % After two delays.
  \draw[axisblue, thick] (5.0,\ysig) -- (5.0,\ysum);
  \draw[axisblue, thick, ->] (5.0,\ysum) -- (5.7,\ysum);
  \node[axisblue, anchor=east] at (4.9,\ysum+0.6) {$\frac{1}{3}$};

\end{circuitikz}
