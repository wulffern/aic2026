% Current starved ring oscillator: a PMOS mirror copies I_control into
% the supply of each inverter, and a Vcontrol PMOS in parallel with each
% mirror device adds a voltage controlled current. Three inverters in a
% ring, each loaded by a capacitor C.
%
% Wires that cross without a dot are not connected.

\begin{circuitikz}[american, thick, transform shape, circuit ee IEC]

  \input{tikz/ckt_lib.tex}

  \def\ydr{5.8}   % drains of the mirror row
  \def\ygr{5.3}   % mirror gate rail
  \def\yvj{2.3}   % where the Vcontrol devices join the supply wires
  \def\yvr{1.7}   % Vcontrol gate rail
  \def\xz{-2.6}   % the diode connected reference

  % ---- PMOS mirror row: diode reference and three outputs ----
  \draw (\xz,\ydr) to [Tpmos, n=P0, mirror] (\xz,{\ydr+1.6});
  \draw (\xz,{\ydr+1.6}) \vsupply;
  \draw (0,\ydr)   to [Tpmos, n=P1] (0,{\ydr+1.6});
  \draw (0,{\ydr+1.6}) \vsupply;
  \draw (2.4,\ydr) to [Tpmos, n=P2] (2.4,{\ydr+1.6});
  \draw (2.4,{\ydr+1.6}) \vsupply;
  \draw (4.8,\ydr) to [Tpmos, n=P3] (4.8,{\ydr+1.6});
  \draw (4.8,{\ydr+1.6}) \vsupply;

  % gate rail below the row, tied to the reference drain
  \draw (P0.G) -- ++(0.2,0) coordinate (t0) -- (t0 |- 0,\ygr) coordinate (gr0);
  \draw (P1.G) -- ++(-0.2,0) coordinate (t1) -- (t1 |- 0,\ygr) coordinate (gr1);
  \draw (P2.G) -- ++(-0.2,0) coordinate (t2) -- (t2 |- 0,\ygr) coordinate (gr2);
  \draw (P3.G) -- ++(-0.2,0) coordinate (t3) -- (t3 |- 0,\ygr) coordinate (gr3);
  \draw (\xz,\ygr) -- (gr3);
  \fill (\xz,\ygr) circle (0.07);
  \fill (gr0) circle (0.07);
  \fill (gr1) circle (0.07);
  \fill (gr2) circle (0.07);

  % reference current
  \draw (\xz,\ydr) -- (\xz,4.6) to [I, l_=$I_{control}$] (\xz,2.8)
        -- (\xz,2.5) \vground;

  % ---- Vcontrol PMOS in parallel with each mirror output ----
  \draw (-1.5,2.9) to [Tpmos, n=V1] (-1.5,4.5);
  \draw (-1.5,4.5) \vsupply;
  \draw (0.9,2.9)  to [Tpmos, n=V2] (0.9,4.5);
  \draw (0.9,4.5) \vsupply;
  \draw (3.3,2.9)  to [Tpmos, n=V3] (3.3,4.5);
  \draw (3.3,4.5) \vsupply;

  % their drains join the inverter supply wires
  \draw (-1.5,2.9) -- (-1.5,\yvj) -- (0,\yvj);
  \fill (0,\yvj) circle (0.07);
  \draw (0.9,2.9) -- (0.9,\yvj) -- (2.4,\yvj);
  \fill (2.4,\yvj) circle (0.07);
  \draw (3.3,2.9) -- (3.3,\yvj) -- (4.8,\yvj);
  \fill (4.8,\yvj) circle (0.07);

  % Vcontrol gate rail
  \draw (V1.G) -- ++(-0.2,0) coordinate (u1) -- (u1 |- 0,\yvr) coordinate (vr1);
  \draw (V2.G) -- ++(-0.2,0) coordinate (u2) -- (u2 |- 0,\yvr) coordinate (vr2);
  \draw (V3.G) -- ++(-0.2,0) coordinate (u3) -- (u3 |- 0,\yvr) coordinate (vr3);
  \draw (-3.6,\yvr) to [short,o-] (vr3);
  \node[anchor=east] at (-3.7,\yvr) {$V_{control}$};
  \fill (vr1) circle (0.07);
  \fill (vr2) circle (0.07);

  % ---- The ring ----
  \node[not port] (g1) at (0,0) {};
  \node[not port] (g2) at (2.4,0) {};
  \node[not port] (g3) at (4.8,0) {};

  \draw (g1.out) -- (g2.in);
  \draw (g2.out) -- (g3.in);
  \draw (g3.out) -- (6.6,0) -- (6.6,-3.0) -- (-1.4,-3.0) -- (-1.4,0)
        -- (g1.in);

  % starved supplies down onto each inverter
  \draw (0,\ydr) -- (g1.north);
  \draw (2.4,\ydr) -- (g2.north);
  \draw (4.8,\ydr) -- (g3.north);

  % load capacitors
  \draw (1.4,0) to [short,*-] (1.4,-0.6);
  \draw (1.4,-2.2) \vground \vcapacitor{$C$};
  \draw (3.8,0) to [short,*-] (3.8,-0.6);
  \draw (3.8,-2.2) \vground \vcapacitor{$C$};
  \draw (6.05,0) to [short,*-] (6.05,-0.6);
  \draw (6.05,-2.2) \vground \vcapacitor{$C$};

\end{circuitikz}
\end{document}
