% Capacitive boost DC/DC (charge pump): charge two capacitors in
% parallel to Vin, then reconfigure them in series to get 2 Vin. The
% letters mark the capacitor terminals through the reconfiguration.

\begin{circuitikz}[american, thick, transform shape, circuit ee IEC]

  \input{tikz/ckt_lib.tex}

  % parallel pair charged to Vin
  \draw (0,4.2) -- (1.6,4.2);
  \node[anchor=south] at (0.8,4.35) {$V_{in}$};
  \draw (0,2.6) \vcapacitor{};
  \draw (0,2.6) \vground;
  \draw (1.6,2.6) \vcapacitor{};
  \draw (1.6,2.6) \vground;

  \node[scale=0.8, anchor=west] at (0.15,3.95) {A};
  \node[scale=0.8, anchor=west] at (0.15,2.90) {B};
  \node[scale=0.8, anchor=west] at (1.75,3.95) {C};
  \node[scale=0.8, anchor=west] at (1.75,2.90) {D};

  % reconfigure
  \draw[->, very thick] (2.9,3.0) -- (4.1,3.0);

  % series stack giving 2 Vin
  \draw (5.3,4.6) -- (5.3,4.8) node[anchor=south] {$+$};
  \node[anchor=west] at (5.9,4.8) {$2\,V_{in}$};
  \draw (5.3,3.0) \vcapacitor{$V_{in}$};
  \draw (5.3,1.4) \vcapacitor{$V_{in}$};
  \draw (5.3,1.4) \vground;
  \node[anchor=south west] at (5.4,1.05) {$-$};

  % terminal letters, top to bottom
  \node[scale=0.8, anchor=east] at (5.15,4.35) {A};
  \node[scale=0.8, anchor=east] at (5.15,3.30) {B};
  \node[scale=0.8, anchor=east] at (5.15,2.75) {C};
  \node[scale=0.8, anchor=east] at (5.15,1.70) {D};

\end{circuitikz}
