% Where the transistor capacitances live. Blue: source diffusion and the
% inversion channel tapering towards the drain. Red: the p- bulk. The
% four capacitances are drawn in green at the junctions they belong to,
% with their labels outside the device.

\definecolor{echarge}{RGB}{30,60,220}
\definecolor{hcharge}{RGB}{220,30,30}
\definecolor{capgreen}{RGB}{20,140,55}

\begin{tikzpicture}[scale=0.9]

  % p- bulk: one wavy blob that crests towards the surface in the
  % pinched gap before the drain, but never touches the diffusions or
  % the channel — the white space in between is the depletion region
  \fill[hcharge!50]
    (-0.2,-3.5) -- (-0.2,-2.4)
    .. controls (0.8,-2.15) and (2.4,-2.5) .. (3.6,-2.1)
    .. controls (4.8,-1.65) and (5.2,-1.05) .. (5.9,-1.0)
    .. controls (6.6,-1.0) and (6.8,-1.7) .. (7.8,-1.85)
    .. controls (8.6,-1.95) and (9.4,-1.6) .. (10.2,-1.8)
    -- (10.2,-3.5) -- cycle;

  % source diffusion + channel wedge (blue)
  \fill[echarge!75, rounded corners=6pt]
    (0.2,0) -- (0.2,-1.5) -- (2.6,-1.5)
    .. controls (3.4,-1.45) and (3.6,-1.0) .. (4.2,-0.5)
    .. controls (5.2,-0.14) and (6.0,-0.07) .. (6.7,-0.05)
    -- (6.7,0) -- cycle;

  % drain n+ region
  \draw[line width=1.2pt, rounded corners=4pt] (7.15,0) -- (7.15,-0.95) -- (9.8,-0.95) -- (9.8,0);
  \node[anchor=center, scale=0.9] at (8.9,-0.45) {$n+$};
  \node[anchor=center, scale=0.9, white] at (1.0,-0.7) {$n+$};

  % surface and gate
  \draw[line width=1.4pt] (-0.2,0) -- (10.2,0);
  \draw[line width=1.6pt, rounded corners=8pt] (2.8,0.18) rectangle (7.2,1.1);
  \node[anchor=center, black!60] at (5.0,0.64) {gate};

  % terminals
  \draw[thick] (1.5,0) -- (1.5,0.75);
  \node[anchor=south] at (1.5,0.8) {$v_s$};
  \draw[thick] (4.0,1.1) -- (4.0,1.6);
  \node[anchor=south] at (4.0,1.65) {$v_g$};
  \draw[thick] (8.5,0) -- (8.5,0.75);
  \node[anchor=south] at (8.5,0.8) {$v_o$};

  % bias labels, top right, clear of everything
  \node[echarge, anchor=west] at (6.3,2.05) {$V_{gs} > V_t$ \quad $V_{DS} > V_{eff}$};

  % green capacitors, labels pulled outside the device
  \newcommand{\gcap}[3]{% x ytop bottomlead, plates centred on the crossing
    \draw[capgreen, thick] (#1,#2) -- (#1,#2-0.3);
    \draw[capgreen, line width=1.7pt] (#1-0.3,#2-0.3) -- (#1+0.3,#2-0.3);
    \draw[capgreen, line width=1.7pt] (#1-0.3,#2-0.48) -- (#1+0.3,#2-0.48);
    \draw[capgreen, thick] (#1,#2-0.48) -- (#1,#2-0.48-#3);
    \fill[capgreen] (#1,#2) circle (0.06);
    \fill[capgreen] (#1,#2-0.48-#3) circle (0.06);
  }

  % C_gs: gate to channel, left end of the gate
  \gcap{3.3}{0.55}{0.3}
  \node[capgreen, anchor=east] at (2.4,1.5) {$C_{gs}$};
  \draw[capgreen, thin] (2.45,1.42) -- (3.15,0.6);

  % C_gd: gate to drain, right end of the gate
  \gcap{6.9}{0.55}{0.3}
  \node[capgreen, anchor=west] at (7.7,1.5) {$C_{gd}$};
  \draw[capgreen, thin] (7.65,1.42) -- (7.05,0.6);

  % C_sb: source to bulk, across the depletion gap
  \gcap{1.7}{-1.2}{0.85}
  \node[capgreen, anchor=east] at (0.9,-2.9) {$C_{sb}$};
  \draw[capgreen, thin] (0.95,-2.82) -- (1.6,-2.4);

  % C_db: drain to bulk, across the depletion gap
  \gcap{8.6}{-0.7}{0.75}
  \node[capgreen, anchor=west] at (9.4,-2.6) {$C_{db}$};
  \draw[capgreen, thin] (9.35,-2.52) -- (8.7,-1.95);

\end{tikzpicture}
\end{document}
