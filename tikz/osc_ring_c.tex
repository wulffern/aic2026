% Ring oscillator with an explicit load capacitor C on each stage to
% control (lower) the frequency. A fourth inverter buffers the output.

\begin{circuitikz}[american, thick, circuit ee IEC]

  \input{tikz/ckt_lib.tex}

  \node[not port] (g1) at (0,0) {};
  \node[not port] (g2) at (2.4,0) {};
  \node[not port] (g3) at (4.8,0) {};
  \node[not port] (g4) at (7.3,0) {};

  \draw (g1.out) -- (g2.in);
  \draw (g2.out) -- (g3.in);
  \draw (g3.out) to [short,-*] (6.05,0) -- (g4.in);
  \draw (g4.out) -- ++(0.5,0);

  % feedback over the top
  \draw (6.05,0) -- (6.05,1.3) -- (-1.4,1.3) -- (-1.4,0) -- (g1.in);

  % load capacitors
  \draw (1.4,0) to [short,*-] (1.4,-0.6);
  \draw (1.4,-2.2) \vground \vcapacitor{$C$};
  \draw (3.8,0) to [short,*-] (3.8,-0.6);
  \draw (3.8,-2.2) \vground \vcapacitor{$C$};
  \draw (6.05,0) -- (6.05,-0.6);
  \draw (6.05,-2.2) \vground \vcapacitor{$C$};

\end{circuitikz}
