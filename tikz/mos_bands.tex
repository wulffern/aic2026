% Conduction band along the MOSFET at V_DS = 0: the bulk under the gate
% is an energy barrier phi_B, and the four injection currents at the two
% junctions sum to zero.

\definecolor{regg}{RGB}{70,130,60}
\definecolor{lblred}{RGB}{160,25,35}
\definecolor{eblue}{RGB}{85,155,235}

\begin{tikzpicture}[scale=0.85]

  \draw[densely dotted, thick, blue!60] (3.3,0.8) -- (3.3,3.3);
  \draw[densely dotted, thick, blue!60] (6.7,0.8) -- (6.7,3.3);

  % conduction band
  \draw[black, thick]
    (0,1.2) -- (2.6,1.2) .. controls (3.3,1.2) and (3.3,2.65) .. (4.0,2.65)
    -- (6.0,2.65) .. controls (6.7,2.65) and (6.7,1.2) .. (7.4,1.2) -- (10,1.2);

  % barrier height
  \draw[black, dashed, thin] (2.3,2.65) -- (4.0,2.65);
  \draw[eblue, thick, <->] (2.55,1.25) -- (2.55,2.65);
  \node[eblue, anchor=east, scale=0.9] at (2.4,1.95) {$\phi_B$};

  % region labels
  \node[lblred, anchor=south, scale=1.1] at (1.5,3.4) {Source};
  \node[lblred, anchor=south, scale=1.1] at (5.0,3.4) {Gate};
  \node[lblred, anchor=south, scale=1.1] at (8.5,3.4) {Drain};
  \node[regg, anchor=center, scale=1.0] at (1.5,0.5) {$n$};
  \node[regg, anchor=center, scale=1.0] at (5.0,0.5) {$p$};
  \node[regg, anchor=center, scale=1.0] at (8.5,0.5) {$n$};

  % the four injection currents
  \node[regg, anchor=center, scale=0.85, circle, draw=regg, inner sep=1.5pt] at (2.6,-0.25) {1};
  \draw[eblue, thick, ->] (2.95,-0.25) -- (3.45,-0.25);
  \node[eblue, anchor=west, scale=0.9] at (3.5,-0.25) {$e^-$};
  \node[eblue, anchor=east, scale=0.9] at (2.35,-0.85) {$e^-$};
  \draw[eblue, thick, ->] (3.0,-0.85) -- (2.5,-0.85);
  \node[regg, anchor=center, scale=0.85, circle, draw=regg, inner sep=1.5pt] at (3.35,-0.85) {2};

  \node[regg, anchor=center, scale=0.85, circle, draw=regg, inner sep=1.5pt] at (6.4,-0.25) {3};
  \draw[eblue, thick, ->] (6.75,-0.25) -- (7.25,-0.25);
  \node[eblue, anchor=west, scale=0.9] at (7.3,-0.25) {$e^-$};
  \node[eblue, anchor=east, scale=0.9] at (6.15,-0.85) {$e^-$};
  \draw[eblue, thick, ->] (6.8,-0.85) -- (6.3,-0.85);
  \node[regg, anchor=center, scale=0.85, circle, draw=regg, inner sep=1.5pt] at (7.15,-0.85) {4};

\end{tikzpicture}
\end{document}
